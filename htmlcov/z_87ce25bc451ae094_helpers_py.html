92def load_gene_list(file_path: str) -> set[str]: 

93 """ 

94 Load genes from a file (one gene per line) into a set of uppercase gene names. 

95 

96 Parameters 

97 ---------- 

98 file_path : str 

99 Path to a file containing gene names, one per line 

100 

101 Returns 

102 ------- 

103 Set[str] 

104 A set of gene names in uppercase for case-insensitive matching 

105 """ 

106 genes: set[str] = set() 

107 if not os.path.exists(file_path): 

108 logger.warning(f"Gene list file not found: {file_path}. Skipping this list.") 

109 return genes 

110 

111 try: 

112 with open(file_path, encoding="utf-8") as f: 

113 for line_num, line_content in enumerate(f, 1): 

114 gene = line_content.strip() 

115 if gene: # Avoid empty lines 

116 if ( 

117 not gene.isalnum() and "_" not in gene and "-" not in gene 

118 ): # Basic check for valid gene name characters 

119 logger.debug( 

120 f"Potentially non-standard character in gene '{gene}' from file {file_path} at line {line_num}. Including anyway." 

121 ) 

122 genes.add(gene.upper()) # Store as uppercase for case-insensitive matching 

123 except Exception as e: 

124 logger.error(f"Error reading gene list file {file_path}: {e}") 

125 

126 if not genes: 

127 logger.warning(f"Gene list file {file_path} was empty or no valid genes found.") 

128 else: 

129 logger.debug(f"Loaded {len(genes)} genes from {file_path}") 

130 

131 return genes

134def _sanitize_column_name(file_path: str) -> str: 

135 """ 

136 Convert a file path to a valid TSV column name. 

137 

138 Extract the filename and remove invalid characters. 

139 

140 Parameters 

141 ---------- 

142 file_path : str 

143 Path to a file 

144 

145 Returns 

146 ------- 

147 str 

148 A sanitized column name derived from the file's basename 

149 """ 

150 base_name = os.path.basename(file_path) 

151 name_part, _ext = os.path.splitext(base_name) 

152 

153 # Iteratively remove common extensions 

154 common_extensions = [".txt", ".list", ".tsv", ".csv", ".genes", ".gene"] 

155 original_name_part = name_part 

156 while True: 

157 found_ext = False 

158 for common_ext in common_extensions: 

159 if name_part.lower().endswith(common_ext): 

160 name_part = name_part[: -len(common_ext)] 

161 found_ext = True 

162 break 

163 if not found_ext: 

164 break 

165 

166 if not name_part: # If all was extension 

167 name_part = original_name_part # Fallback to name before extension stripping 

168 

169 # Replace non-alphanumeric characters (except underscore) with underscore 

170 sanitized_name = re.sub(r"[^\w_]", "_", name_part) 

171 # Remove leading/trailing underscores and collapse multiple underscores 

172 sanitized_name = re.sub(r"_+", "_", sanitized_name).strip("_") 

173 

174 if not sanitized_name: # If name becomes empty after sanitization (e.g. "---.txt") 

175 sanitized_name = "unnamed_list" 

176 

177 # Ensure it doesn't start with a number or is purely numeric 

178 if not sanitized_name or sanitized_name[0].isdigit() or sanitized_name.isdigit(): 

179 sanitized_name = "list_" + sanitized_name 

180 

181 return sanitized_name

184def annotate_variants_with_gene_lists(lines: list[str], gene_list_files: list[str]) -> list[str]: 

185 """ 

186 Add new columns to variant data lines indicating membership in provided gene lists. 

187 

188 For each gene list file, a new column is added to the TSV file. The column will contain 'yes' if 

189 any of the genes in the GENE column (which may be comma-separated) is present in the gene list, 

190 and 'no' otherwise. 

191 

192 Parameters 

193 ---------- 

194 lines : List[str] 

195 Lines of the TSV file to annotate 

196 gene_list_files : List[str] 

197 List of file paths containing gene names (one per line) 

198 

199 Returns 

200 ------- 

201 List[str] 

202 The input lines with additional columns for each gene list 

203 

204 Notes 

205 ----- 

206 - If the GENE column is missing, an error is logged and the input lines are returned unchanged 

207 - Multiple genes in the GENE column can be separated by commas, semicolons, or spaces 

208 - If a gene list file fails to load, it is skipped with an error message 

209 - Column names are sanitized from the file basename, duplicates are suffixed with numbers 

210 """ 

211 if not gene_list_files or not lines or len(lines) < 2: 

212 return lines 

213 

214 loaded_gene_sets: dict[str, set[str]] = {} 

215 ordered_new_column_names: list[str] = [] 

216 

217 for file_path in gene_list_files: 

218 genes = load_gene_list(file_path) 

219 if genes: # Only process if genes were successfully loaded 

220 base_col_name = _sanitize_column_name(file_path) 

221 final_col_name = base_col_name 

222 suffix = 1 

223 # Ensure unique column names if multiple files sanitize to the same name 

224 while final_col_name in loaded_gene_sets: 

225 final_col_name = f"{base_col_name}_{suffix}" 

226 suffix += 1 

227 

228 loaded_gene_sets[final_col_name] = genes 

229 ordered_new_column_names.append(final_col_name) 

230 logger.info( 

231 f"Loaded {len(genes)} genes for annotation column '{final_col_name}' from {file_path}" 

232 ) 

233 

234 if not loaded_gene_sets: 

235 logger.warning("No valid gene lists were loaded; no annotation columns will be added.") 

236 return lines 

237 

238 # Process header 

239 header_line = lines[0].rstrip("\n") 

240 header_parts = header_line.split("\t") 

241 

242 try: 

243 gene_col_idx = header_parts.index("GENE") 

244 except ValueError: 

245 logger.error( 

246 "CRITICAL: 'GENE' column not found in TSV header. Cannot perform gene list annotation." 

247 ) 

248 # Return lines unmodified to prevent pipeline breakage if GENE column is unexpectedly missing 

249 return lines 

250 

251 new_header_string = "\t".join(header_parts + ordered_new_column_names) 

252 modified_lines = [new_header_string] 

253 

254 # Process data lines 

255 for data_line_str in lines[1:]: 

256 data_line_str = data_line_str.rstrip("\n") 

257 if not data_line_str.strip(): # Preserve empty lines if any 

258 modified_lines.append(data_line_str) 

259 continue 

260 

261 fields = data_line_str.split("\t") 

262 

263 # Ensure fields list is long enough for gene_col_idx 

264 if gene_col_idx >= len(fields): 

265 logger.warning( 

266 f"Line has too few fields to access GENE column (index {gene_col_idx}): {data_line_str}" 

267 ) 

268 # Append "no" for all new columns for this malformed line 

269 fields.extend(["" for _ in range(gene_col_idx - len(fields) + 1)]) 

270 fields.extend(["no" for _ in ordered_new_column_names]) 

271 modified_lines.append("\t".join(fields)) 

272 continue 

273 

274 variant_gene_field_value = fields[gene_col_idx].strip().upper() # Match in uppercase 

275 

276 # Handle if GENE column contains multiple genes (e.g., comma-separated or other delimiters) 

277 if variant_gene_field_value: 

278 variant_genes_in_record = { 

279 g.strip() for g in re.split(r"[,; ]+", variant_gene_field_value) if g.strip() 

280 } 

281 else: 

282 variant_genes_in_record = set() 

283 

284 for col_name in ordered_new_column_names: 

285 gene_set_for_current_list = loaded_gene_sets[col_name] 

286 is_member = "no" 

287 # Check for intersection between variant genes and gene list 

288 if variant_genes_in_record and not variant_genes_in_record.isdisjoint( 

289 gene_set_for_current_list 

290 ): 

291 is_member = "yes" 

292 fields.append(is_member) 

293 

294 modified_lines.append("\t".join(fields)) 

295 

296 logger.info( 

297 f"Annotation with gene lists complete. Added {len(ordered_new_column_names)} columns." 

298 ) 

299 return modified_lines

351def get_vcf_names(vcf_file: str) -> list[str]: 

352 """ 

353 Get sample names from a VCF file header. 

354 

355 Parameters 

356 ---------- 

357 vcf_file : str 

358 Path to the VCF file 

359 

360 Returns 

361 ------- 

362 List[str] 

363 List of sample names from the VCF file 

364 """ 

365 try: 

366 from subprocess import PIPE, Popen 

367 

368 cmd = ["bcftools", "query", "-l", vcf_file] 

369 process = Popen(cmd, stdout=PIPE, stderr=PIPE, universal_newlines=True) 

370 stdout, stderr = process.communicate() 

371 

372 if process.returncode != 0: 

373 logger.error(f"Error getting sample names from VCF: {stderr}") 

374 raise RuntimeError(f"bcftools query -l failed for {vcf_file}: {stderr.strip()}") 

375 

376 return [line.strip() for line in stdout.split("\n") if line.strip()] 

377 except RuntimeError: 

378 raise 

379 except Exception as e: 

380 logger.error(f"Error running bcftools query: {e}") 

381 raise RuntimeError(f"Error running bcftools query on {vcf_file}: {e}") from e

384def get_vcf_regions(vcf_file: str) -> list[str]: 

385 """ 

386 Get the chromosomal regions covered in a VCF file. 

387 

388 Parameters 

389 ---------- 

390 vcf_file : str 

391 Path to the VCF file 

392 

393 Returns 

394 ------- 

395 List[str] 

396 List of regions in format "chr:start-end" 

397 """ 

398 try: 

399 from subprocess import PIPE, Popen 

400 

401 cmd = ["bcftools", "query", "-f", "%CHROM:%POS\n", vcf_file] 

402 process = Popen(cmd, stdout=PIPE, stderr=PIPE, universal_newlines=True) 

403 stdout, stderr = process.communicate() 

404 

405 if process.returncode != 0: 

406 logger.error(f"Error getting regions from VCF: {stderr}") 

407 return [] 

408 

409 positions = [line.strip() for line in stdout.split("\n") if line.strip()] 

410 if not positions: 

411 return [] 

412 

413 # Group by chromosome and find min/max positions 

414 by_chrom = defaultdict(list) 

415 for pos in positions: 

416 if ":" in pos: 

417 chrom, position = pos.split(":", 1) 

418 by_chrom[chrom].append(int(position)) 

419 

420 return [ 

421 f"{chrom}:{min(positions)}-{max(positions)}" for chrom, positions in by_chrom.items() 

422 ] 

423 except Exception as e: 

424 logger.error(f"Error analyzing VCF regions: {e}") 

425 return []

458 """

468 )

475 )

477 if bcf_proc.stdout: 

478 bcf_proc.stdout.close() 

479 

480 stdout, _stderr = wc_proc.communicate() 

481 bcf_proc.wait() 

482 

483 if bcf_proc.returncode != 0: 

484 bcf_stderr = bcf_proc.stderr.read().decode() if bcf_proc.stderr else "" 

485 logger.error(f"Error getting VCF size: {bcf_stderr}") 

486 return 0 

487 

488 try: 

489 return int(stdout.strip()) 

490 except ValueError: 

491 logger.error(f"Could not parse VCF size output: {stdout}") 

492 return 0 

493 except Exception as e: 

494 logger.error(f"Error counting variants in VCF: {e}") 

495 return 0

498def match_igv_link_columns(header_fields: list[str]) -> dict[str, int]: 

499 """ 

500 Find columns in a TSV header that should contain IGV links. 

501 

502 Parameters 

503 ---------- 

504 header_fields : List[str] 

505 List of header field names 

506 

507 Returns 

508 ------- 

509 Dict[str, int]

511 """ 

512 # Columns that might contain coordinates for IGV links 

513 igv_column_patterns = ["CHROM", "POS", "START", "END", "POSITION", "LOCATION"]

523def read_sequencing_manifest(file_path: str) -> dict[str, dict[str, str]]: 

524 """ 

525 Read a sequencing manifest file containing sample metadata. 

526 

527 Expected format: TSV with header row and sample IDs in first column. 

528 

529 Parameters 

530 ---------- 

531 file_path : str 

532 Path to the manifest file 

533 

534 Returns 

535 ------- 

536 Dict[str, Dict[str, str]] 

537 Mapping of sample IDs to metadata dictionaries 

538 """ 

539 result: dict[str, dict[str, Any]] = {} 

540 if not os.path.exists(file_path): 

541 logger.warning(f"Manifest file not found: {file_path}") 

542 return result 

543 

544 try: 

545 df = pd.read_csv(file_path, sep="\t", dtype=str) 

546 if df.empty: 

547 logger.warning(f"Empty manifest file: {file_path}") 

548 return result 

549 

550 # Assuming first column contains sample IDs 

551 id_col = df.columns[0] 

552 

553 # Convert each row to a dictionary 

554 for row in df.itertuples(index=True): 

555 sample_id = getattr(row, id_col, None) 

556 if sample_id and not pd.isna(sample_id): 

557 # Convert namedtuple to dict for metadata storage 

558 result[str(sample_id)] = cast(dict[str, Any], df.loc[row.Index].to_dict()) 

559 

560 logger.info(f"Loaded metadata for {len(result)} samples from {file_path}") 

561 return result 

562 except Exception as e: 

563 logger.error(f"Error reading sequencing manifest: {e}") 

564 return result
