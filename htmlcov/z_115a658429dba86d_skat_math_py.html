37def compute_eigenvalues_filtered( 

38 geno_weighted: np.ndarray, 

39 null_model: NullModelResult, 

40) -> np.ndarray: 

41 """ 

42 Compute filtered eigenvalues of the null-projected SKAT kernel ``Z'PZ / 2``. 

43 

44 ``P`` is the null-model score-covariance projector (see 

45 ``project_out_nuisance``): Bernoulli-variance weighted for binary traits, 

46 ``I - X(X'X)^-1 X'`` for quantitative traits. For quantitative traits the 

47 matching statistic is ``Q = ||Z'r||^2 / (2 sigma2)`` (R ``SKAT.linear``), 

48 so the eigenvalues carry no ``sigma2`` and are invariant to outcome units. 

49 

50 Uses scipy.linalg.eigh(driver='evr') matching R's DSYEVR Lapack routine. 

51 Applies threshold = mean(positive_eigenvalues) / 100_000 (matches R SKAT), 

52 floored at ``projection_noise_floor(...)**2 / 2`` (round-off of ``Z'PZ``). 

53 

54 Parameters 

55 ---------- 

56 geno_weighted : np.ndarray, shape (n_samples, n_variants) 

57 Weighted genotype matrix (G * W). 

58 null_model : NullModelResult 

59 Fitted null model containing mu_hat (binary) and design_matrix. 

60 

61 Returns 

62 ------- 

63 np.ndarray 

64 Filtered eigenvalues (positive and above threshold). Empty array 

65 if none pass. 

66 """ 

67 from variantcentrifuge.association.backends.null_model import ( 

68 project_out_nuisance, 

69 projection_noise_floor, 

70 ) 

71 

72 z_adj = project_out_nuisance(geno_weighted, null_model) 

73 n_variants = geno_weighted.shape[1] 

74 eig_mat = z_adj.T @ z_adj if n_variants <= geno_weighted.shape[0] else z_adj @ z_adj.T 

75 lambdas_all = scipy.linalg.eigh(eig_mat / 2.0, eigvals_only=True, driver="evr") 

76 

77 pos = lambdas_all[lambdas_all >= 0] 

78 if len(pos) == 0: 

79 return np.array([], dtype=np.float64) 

80 # R's relative filter, floored at the round-off level of the unprojected kernel 

81 # (directions explained by covariates must not become noise eigenvalues). 

82 threshold = max( 

83 pos.mean() / 100_000.0, projection_noise_floor(geno_weighted, null_model) ** 2 / 2.0 

84 ) 

85 return np.asarray(lambdas_all[lambdas_all > threshold], dtype=np.float64)

88def compute_skato_q_and_z1( 

89 geno_weighted: np.ndarray, 

90 null_model: NullModelResult, 

91 rho_grid: list[float], 

92) -> tuple[np.ndarray, np.ndarray]: 

93 """Compute per-rho Q statistics and projection-adjusted Z1/sqrt(2) for SKAT-O.""" 

94 from variantcentrifuge.association.backends.null_model import project_out_nuisance 

95 

96 residuals: np.ndarray = null_model.extra["residuals"] 

97 # Binary sigma2 is 1; quantitative Q is scaled by the residual variance (R SKAT_Optimal_Linear) 

98 sigma2: float = null_model.extra["sigma2"] 

99 

100 # ── Step 1: Compute Q_rho for each rho ── 

101 score_vec = geno_weighted.T @ residuals 

102 score_sum = float(np.sum(score_vec)) 

103 

104 q_rho = np.zeros(len(rho_grid), dtype=np.float64) 

105 for i, rho in enumerate(rho_grid): 

106 q1 = (1.0 - rho) * float(score_vec @ score_vec) 

107 q2 = rho * score_sum**2 

108 q_rho[i] = (q1 + q2) / (2.0 * sigma2) 

109 

110 # ── Step 2: Build projection-adjusted Z1 ── 

111 z1_half = project_out_nuisance(geno_weighted, null_model) / np.sqrt(2.0) 

112 return q_rho, z1_half
