1""" 

2Imbalance-robust binary score tests: saddlepoint (SPA) and exact (ER) kernel. 

3 

4Port of R ``SKAT::SKATBinary_Robust`` (SKAT 2.2.5, Zhao et al. 2020 AJHG) with 

5``SPAtest::Saddle_Prob`` (3.1.2), vectorised over the variants of a gene: 

6 

7- per variant, the score of the covariate-projected genotype ``G1`` is tested with 

8 the normal approximation; where ``|z| > 2`` the p-value is replaced by a 

9 saddlepoint p-value (MAC >= 10) or an exact carrier-enumeration p-value (MAC < 10); 

10- the calibrated variance ``V~_j = S_j^2 / qchisq(1 - p~_j, 1)`` rescales the score 

11 covariance, the burden direction is recalibrated by an SPA burden p-value, and 

12 SKAT / SKAT-O are evaluated on the calibrated (score, covariance) pair. 

13 

14``variant_pvalues`` is the per-variant kernel (reused for score artifacts); 

15``robust_skat`` is the gene-level SKAT / SKAT-O p-value. 

16"""

27_CUTOFF = 2.0 # |z| above which SPA / ER replace the normal approximation 

28_ER_MAC = 10 # R: exact p-value when colSums(G) < 10 

29_TOL = np.finfo(np.float64).eps ** 0.25 # SPAtest getroot_K1 tolerance 

30_MAXITER = 1000 

31_ER_EPSILON = 1e-6 # SKATBinary tie tolerance on the exact test statistic

34def _k1(t: np.ndarray, mu: np.ndarray, g: np.ndarray, q: np.ndarray) -> np.ndarray: 

35 """First CGF derivative minus q (SPAtest ``K1_adj``), one column of g per t.""" 

36 mu_c = mu[:, np.newaxis] 

37 return np.asarray(np.sum(mu_c * g / ((1.0 - mu_c) * np.exp(-g * t) + mu_c), axis=0) - q)

40def _k2(t: np.ndarray, mu: np.ndarray, g: np.ndarray) -> np.ndarray: 

41 """Second CGF derivative (SPAtest ``K2``, ``na.rm = TRUE``).""" 

42 mu_c = mu[:, np.newaxis] 

43 e = np.exp(-g * t) 

44 return np.asarray( 

45 np.nansum((1.0 - mu_c) * mu_c * g**2 * e / ((1.0 - mu_c) * e + mu_c) ** 2, axis=0) 

46 )

49def _korg(t: np.ndarray, mu: np.ndarray, g: np.ndarray) -> np.ndarray: 

50 """Cumulant generating function (SPAtest ``Korg``).""" 

51 mu_c = mu[:, np.newaxis] 

52 return np.asarray(np.sum(np.log(1.0 - mu_c + mu_c * np.exp(g * t)), axis=0))

55def _root(mu: np.ndarray, g: np.ndarray, q: np.ndarray) -> tuple[np.ndarray, np.ndarray]: 

56 """Saddlepoint roots for all columns at once: SPAtest ``getroot_K1`` iterate-for-iterate.""" 

57 gpos = np.where(g > 0, g, 0.0).sum(axis=0) 

58 gneg = np.where(g < 0, g, 0.0).sum(axis=0) 

59 outside = (q >= gpos) | (q <= gneg) 

60 t = np.where(outside, np.inf, 0.0) 

61 conv = np.ones(q.size, dtype=bool) 

62 jump = np.full(q.size, np.inf) 

63 act = np.flatnonzero(~outside) 

64 k1 = _k1(t[act], mu, g[:, act], q[act]) 

65 rep = 1 

66 while act.size: 

67 t_act = t[act] 

68 tnew = t_act - k1 / _k2(t_act, mu, g[:, act]) 

69 nan = np.isnan(tnew) 

70 small = ~nan & (np.abs(tnew - t_act) < _TOL) 

71 stop = nan | small | (rep == _MAXITER) 

72 conv[act[stop & ~small]] = False 

73 keep = ~stop 

74 act, t_act, tnew, k1 = act[keep], t_act[keep], tnew[keep], k1[keep] 

75 newk1 = _k1(tnew, mu, g[:, act], q[act]) 

76 flip = np.sign(k1) != np.sign(newk1) 

77 step = np.abs(tnew - t_act) 

78 big = flip & (step > jump[act] - _TOL) 

79 jump[act[flip & ~big]] = step[flip & ~big] 

80 if big.any(): 

81 b = act[big] 

82 tnew[big] = t_act[big] + np.sign(newk1[big] - k1[big]) * jump[b] / 2.0 

83 newk1[big] = _k1(tnew[big], mu, g[:, b], q[b]) 

84 jump[b] /= 2.0 

85 t[act] = tnew 

86 k1 = newk1 

87 rep += 1 

88 return t, conv

91def _tail( 

92 zeta: np.ndarray, mu: np.ndarray, g: np.ndarray, q: np.ndarray, noadj: np.ndarray 

93) -> np.ndarray: 

94 """One-sided saddlepoint tail (SPAtest ``Get_Saddle_Prob``; its error path -> noadj/2).""" 

95 k1 = _korg(zeta, mu, g) 

96 k2 = _k2(zeta, mu, g) 

97 w = np.sign(zeta) * np.sqrt(2.0 * (zeta * q - k1)) 

98 z = w + np.log(zeta * np.sqrt(k2) / w) / w 

99 p = np.where(z > 0, scipy.stats.norm.sf(z), -scipy.stats.norm.cdf(z)) 

100 p = np.where(np.isnan(z), noadj / 2.0, p) 

101 return np.where(np.isfinite(k1) & np.isfinite(k2), p, 0.0)

104def saddle_pvalues(q: np.ndarray, mu: np.ndarray, g: np.ndarray, cutoff: float) -> np.ndarray: 

105 """Two-sided SPA p-values of the scores ``q_j = g_j'y`` (SPAtest ``Saddle_Prob``).""" 

106 m1 = mu @ g 

107 var1 = (mu * (1.0 - mu)) @ g**2 

108 noadj = np.asarray(scipy.stats.chi2.sf((q - m1) ** 2 / var1, 1), dtype=np.float64) 

109 p = noadj.copy() 

110 spa = np.abs(q - m1) / np.sqrt(var1) >= cutoff 

111 if not spa.any(): 

112 return p 

113 gs, qs, na = g[:, spa], q[spa], noadj[spa] 

114 qinv = 2.0 * m1[spa] - qs 

115 with np.errstate(all="ignore"): 

116 r1, c1 = _root(mu, gs, qs) 

117 r2, c2 = _root(mu, gs, qinv) 

118 ps = np.abs(_tail(r1, mu, gs, qs, na)) + np.abs(_tail(r2, mu, gs, qinv, na)) 

119 ps = np.where(c1 & c2, ps, na) 

120 # SPAtest re-runs with a doubled cutoff until it reaches |z|, which ends at the 

121 # normal p-value whenever the saddlepoint p is over 1000x smaller than it 

122 ps = np.where((ps != 0) & (na / ps > 1e3), na, ps) 

123 p[spa] = ps 

124 return p

127def _lchoose(n: int, k: np.ndarray) -> np.ndarray: 

128 """log C(n, k) with the SKAT C++ convention ``k > n -> 0``.""" 

129 with np.errstate(invalid="ignore"): 

130 lc = scipy.special.gammaln(n + 1) - scipy.special.gammaln(k + 1) 

131 lc = lc - scipy.special.gammaln(n - k + 1) 

132 return np.where(k > n, 0.0, np.where(k < 0, -np.inf, lc))

135def exact_pvalue(geno_col: np.ndarray, y: np.ndarray, mu: np.ndarray) -> float: 

136 """Exact single-variant mid-p (SKATBinary ``method.bin = "ER"``) by carrier enumeration.""" 

137 idx = np.flatnonzero(geno_col > 0) 

138 k, n = idx.size, y.size 

139 res = y - mu 

140 # The beta(1, 25) weight only scales the statistic against the 1e-6 tie tolerance 

141 z = geno_col[idx] * scipy.stats.beta.pdf(geno_col.mean() / 2.0, 1, 25) 

142 p1 = mu[idx] 

143 # P(j cases among carriers | total cases): carriers binned into 10 fitted-probability 

144 # groups (SKATExactBin.ComputeProb_Group, type.group = 2, k <= 50), Fisher noncentral 

145 pc = np.minimum(p1, 0.999) 

146 bins = np.searchsorted(np.arange(1, 10) / 10.0, pc, side="right") 

147 p2 = np.delete(mu, idx).mean() 

148 poly = np.ones(1) 

149 for b in np.unique(bins): 

150 size = int(np.sum(bins == b)) 

151 pm = pc[bins == b].mean() 

152 c = np.arange(size + 1) 

153 poly = np.convolve( 

154 poly, scipy.special.comb(size, c) * ((pm / (1 - pm)) / (p2 / (1 - p2))) ** c 

155 ) 

156 ncase = int(np.round(y).sum()) 

157 j = np.arange(k + 1) 

158 with np.errstate(divide="ignore"): 

159 lp = np.log(poly) + _lchoose(n - k, ncase - j) 

160 lp[j > ncase] = -np.inf 

161 prob_k = np.exp(lp - lp.max()) 

162 prob_k /= prob_k.sum() 

163 masks = ((np.arange(2**k)[:, np.newaxis] >> np.arange(k)) & 1).astype(bool) 

164 size = masks.sum(axis=1) 

165 stat = (masks @ z - z @ p1) ** 2 

166 fprob = np.prod(np.where(masks, (p1 / (1 - p1))[np.newaxis, :], 1.0), axis=1) 

167 denom = np.bincount(size, weights=fprob, minlength=k + 1) 

168 fprob = fprob / denom[size] * prob_k[size] 

169 fprob /= fprob.sum() 

170 q_obs = ((res[idx] > 0) @ z - z @ p1) ** 2 

171 d = q_obs - stat 

172 d[np.abs(d) <= _ER_EPSILON] = 0.0 

173 return float(fprob[d <= 0].sum() - fprob[d == 0].sum() / 2.0)

176def variant_pvalues( 

177 geno: np.ndarray, y: np.ndarray, mu: np.ndarray, design: np.ndarray 

178) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]: 

179 """Per-variant robust p-values and calibrated variances (R ``SPA_ER_kernel``). 

180 

181 Columns with MAF > 0.5 are flipped to the minor allele first, as in R. Returns 

182 ``(score, p, g1, v_cal)``: the projected scores ``G1'(y - mu)``, the SPA/ER-calibrated 

183 p-values, the projected genotypes ``G1 = G - X (X'WX)^-1 X'W G`` (all in the flipped 

184 orientation; two-sided p-values are orientation-invariant) and the calibrated 

185 variances ``V~`` (:func:`calibrated_variance`). 

186 

187 Per gene the score artifact and the robust test ask for the same columns under the 

188 same null (``--skat-robust`` with ``--score-artifact``): an input bit-identical to the 

189 previous call returns that call's (read-only) result. 

190 """ 

191 geno = np.where(geno.mean(axis=0) / 2.0 > 0.5, 2.0 - geno, geno) 

192 key = (geno, y, mu, design) 

193 if _last and all(_identical(a, b) for a, b in zip(_last[0], key, strict=True)): 

194 hit: tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray] = _last[1] 

195 return hit 

196 result = _variant_pvalues(geno, y, mu, design) 

197 for arr in result: 

198 arr.flags.writeable = False # shared by the artifact hook and the robust test 

199 _last[:] = [tuple(np.array(a) for a in key), result] # copies: null arrays may change 

200 return result

207def _identical(a: np.ndarray, b: np.ndarray) -> bool: 

208 """Same dtype, shape, layout and bytes (so the same arithmetic gives the same bits).""" 

209 return ( 

210 a.dtype == b.dtype 

211 and a.shape == b.shape 

212 and a.strides == b.strides 

213 and a.tobytes() == b.tobytes() 

214 )

217def _variant_pvalues( 

218 geno: np.ndarray, y: np.ndarray, mu: np.ndarray, design: np.ndarray 

219) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]: 

220 """:func:`variant_pvalues` on minor-allele (already flipped) columns, uncached.""" 

221 xv = design * (mu * (1.0 - mu))[:, np.newaxis] 

222 g1 = geno - design @ np.linalg.solve(design.T @ xv, xv.T @ geno) 

223 mac = geno.sum(axis=0) 

224 g = g1 / np.sqrt(mac) 

225 q = g.T @ y 

226 m1 = mu @ g 

227 var1 = (mu * (1.0 - mu)) @ g**2 

228 stat = (q - m1) ** 2 / var1 

229 p = scipy.stats.chi2.sf(stat, 1) 

230 spa = (stat > _CUTOFF**2) & (mac >= _ER_MAC) 

231 if spa.any(): 

232 ps = saddle_pvalues(q[spa], mu, g[:, spa], _CUTOFF) 

233 p[spa] = np.where(ps != 0, ps, p[spa]) 

234 for col in np.flatnonzero((stat > _CUTOFF**2) & (mac < _ER_MAC)): 

235 p[col] = exact_pvalue(geno[:, col], y, mu) 

236 score = (q - m1) * np.sqrt(mac) 

237 # |z| <= 2 keeps the normal p, where V~ is exactly the score variance: return it 

238 # directly instead of S^2 / qchisq(chi2.sf(z^2)) (round-off at near-zero scores). 

239 v_cal = np.where(stat > _CUTOFF**2, calibrated_variance(score, p), var1 * mac) 

240 return score, p, g1, v_cal

243def calibrated_variance(score: np.ndarray, p: np.ndarray) -> np.ndarray: 

244 """``V~ = S^2 / qchisq(1 - p~, 1)``; ``S^2 / 500`` at p~ = 0, 0 where undefined (p~ = 1).""" 

245 with np.errstate(divide="ignore", invalid="ignore"): 

246 v = np.where(p > 0, score**2 / scipy.stats.chi2.isf(p, 1), score**2 / 500.0) 

247 return np.where(np.isfinite(v), v, 0.0)

257def robust_moments( 

258 geno: np.ndarray, weights: np.ndarray, y: np.ndarray, mu: np.ndarray, design: np.ndarray 

259) -> tuple[np.ndarray, np.ndarray | None, float, np.ndarray]: 

260 """Calibrated unweighted score and covariance of a gene (R ``SKATBinary_spa``). 

261 

262 Returns ``(score, cov, p_burden, p)``; ``diag(w) score`` and ``diag(w) cov diag(w)`` 

263 are R's ``Score`` / ``Phi``. ``cov`` is None where R falls back to the robust 

264 burden p-value ``p_burden`` (degenerate burden recalibration); ``p`` are the 

265 per-variant SPA/ER p-values. 

266 """ 

267 score, p, g1, v_cal = variant_pvalues(geno, y, mu, design) 

268 score = np.where(v_cal > 0, score, 0.0) 

269 cov = g1.T @ (g1 * (mu * (1.0 - mu))[:, np.newaxis]) 

270 d = np.sqrt(v_cal / np.diag(cov)) 

271 cov = cov * np.outer(d, d) 

272 # Every variant enters the burden; R drops p~ = 1 (exactly zero score), which made p 

273 # jump between S = 0 and S ~ 1e-9 (a deliberate departure from R). 

274 g_sum = g1 @ weights 

275 p_burden = float(saddle_pvalues(np.array([g_sum @ y]), mu, g_sum[:, np.newaxis], _CUTOFF)[0]) 

276 # Burden-direction recalibration: shrink the covariance so w'Σw matches the SPA burden 

277 var_q = float(weights @ cov @ weights) 

278 var_b = float((mu * (1.0 - mu)) @ g_sum**2) 

279 if (g_sum @ (y - mu)) ** 2 < _CUTOFF**2 * var_b: 

280 # Normal branch: S^2 / qchisq(1 - p, 1) is exactly var_b; the ratio is round-off 

281 # when opposing variants cancel (S ~ 0, p ~ 1; R's 0/0 at S = 0), as for v_cal in 

282 # variant_pvalues. Departs from R where R's own round-off decides. 

283 var_q2 = var_b 

284 else: 

285 with np.errstate(divide="ignore"): 

286 var_q2 = float(score @ weights) ** 2 / scipy.stats.chi2.isf(p_burden, 1) 

287 r = 1.0 if var_q2 == 0 else min(var_q / var_q2, 1.0) 

288 if not (r > 0 and np.isfinite(r)): 

289 return score, None, p_burden, p 

290 return score, cov / r, p_burden, p

293def robust_skat( 

294 geno: np.ndarray, 

295 weights: np.ndarray, 

296 y: np.ndarray, 

297 mu: np.ndarray, 

298 design: np.ndarray, 

299 method: str, 

300) -> tuple[float | None, float | None, np.ndarray]: 

301 """Robust binary SKAT ("SKAT") or SKAT-O ("SKATO") p-value (R ``SKATBinary_Robust``). 

302 

303 Returns ``(p_value, rho, variant_p)``; ``rho`` is None for SKAT; ``variant_p`` are 

304 the per-variant SPA/ER p-values (for ACAT-V). 

305 """ 

306 score, cov, p_burden, p = robust_moments(geno, weights, y, mu, design) 

307 if cov is None: 

308 return p_burden, None, p # R: Met_SKAT_Get_Pvalue fails -> robust burden only 

309 if method.upper() == "SKATO" and geno.shape[1] > 1: 

310 p_o, rho = meta_skato([score], [cov], weights) 

311 if p_o is not None: 

312 return p_o, rho, p 

313 p_skat = meta_skat([score], [cov], weights)[0] 

314 return p_skat, (0.0 if method.upper() == "SKATO" else None), p
