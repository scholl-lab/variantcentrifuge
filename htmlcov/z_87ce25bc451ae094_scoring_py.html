23def read_scoring_config(config_path: str) -> dict[str, Any]: 

24 """Read and parse scoring configuration files from a directory.""" 

25 try: 

26 logger.debug(f"Reading scoring config from: {config_path}") 

27 p1 = os.path.join(config_path, "variable_assignment_config.json") 

28 p2 = os.path.join(config_path, "formula_config.json") 

29 with open(p1, encoding="utf-8") as f1, open(p2, encoding="utf-8") as f2: 

30 v_cfg, f_cfg = json.load(f1), json.load(f2) 

31 variables = v_cfg.get("variables", {}) if isinstance(v_cfg, dict) else None 

32 formulas = f_cfg.get("formulas", []) if isinstance(f_cfg, dict) else None 

33 if not isinstance(variables, (dict, list)): 

34 raise ValueError("Scoring configuration 'variables' must be a JSON object.") 

35 if not isinstance(formulas, list): 

36 raise ValueError("Scoring configuration 'formulas' must be a JSON array.") 

37 if not all(isinstance(item, dict) for item in formulas): 

38 raise ValueError("Each entry in scoring 'formulas' must be a non-empty object.") 

39 # "_"-prefixed keys are comments, as at the top level of formula_config.json 

40 formulas = [{k: v for k, v in item.items() if not k.startswith("_")} for item in formulas] 

41 for item in formulas: 

42 if not item: 

43 raise ValueError("Each entry in scoring 'formulas' must be a non-empty object.") 

44 for name, expr in item.items(): 

45 if not isinstance(expr, str) or not expr.strip(): 

46 raise ValueError(f"Formula '{name}' must be a non-empty string expression.") 

47 try: 

48 ast.parse(expr, mode="eval") 

49 except SyntaxError as syn_err: 

50 raise ValueError(f"Invalid syntax in formula '{name}': {syn_err}") from syn_err 

51 return { 

52 "variables": variables, 

53 "formulas": formulas, 

54 "output_scores": f_cfg.get("output_scores"), 

55 } 

56 except FileNotFoundError as e: 

57 logger.error(f"Scoring configuration file not found: {e.filename}") 

58 raise 

59 except json.JSONDecodeError as e: 

60 logger.error(f"Error parsing scoring configuration JSON: {e}") 

61 raise

94 def _to_b(x: Any) -> bool: 

95 if pd.isna(x): 

96 return def_b 

97 if isinstance(x, (int, float, np.number, bool, np.bool_)): 

98 return bool(x) 

99 s = str(x).strip().lower() 

100 return True if s in _TRUE_STRS else (False if s in _FALSE_STRS else def_b)

122def _formula_usage(name: str, formulas: list[dict[str, Any]], typed: dict[str, str]) -> set[str]: 

123 """Return how formulas use variable *name* ("numeric", "categorical", ...). 

124 

125 *typed* maps every variable to its explicit or known-column type ("" when untyped). 

126 ponytail: transition guard for rule-based variable typing (only detects disagreement, 

127 never chooses a type); delete in a later release once configs carry explicit types. 

128 """ 

129 norm = lambda s: re.sub(r"\W", "_", s) # noqa: E731 

130 name = norm(name) 

131 

132 def has_var(e: ast.AST, anywhere: bool = False) -> bool: 

133 nodes = ast.walk(e) if anywhere else [e] 

134 return any(isinstance(x, ast.Name) and x.id == name for x in nodes) 

135 

136 def use_of(other: ast.AST, op: ast.AST | None = None) -> str: 

137 if isinstance(other, (ast.List, ast.Tuple, ast.Set)): 

138 uses = {use_of(e) for e in other.elts} 

139 return "categorical" if "categorical" in uses else max(uses, default="") 

140 val = other.value if isinstance(other, ast.Constant) else None 

141 if isinstance(val, str): 

142 return "" if val.strip().lower() in _MISSING_TOKENS else "categorical" 

143 if type(val) in (int, float) or isinstance(op, _ORDERING): 

144 return "numeric" 

145 if isinstance(other, ast.Name) and other.id in typed: 

146 return typed[other.id] or "comparison with an untyped variable" 

147 return "" 

148 

149 used: set[str] = set() 

150 for expr in (str(e) for f in formulas for e in f.values()): 

151 try: 

152 tree = ast.parse(re.sub(r"`([^`]+)`", lambda m: norm(m.group(1)), expr), mode="eval") 

153 except SyntaxError: 

154 continue 

155 for n in ast.walk(tree): 

156 if isinstance(n, ast.BinOp) and not isinstance(n.op, _BITWISE): 

157 used.update(["numeric"] if has_var(n.left) or has_var(n.right) else []) 

158 elif isinstance(n, ast.Compare): 

159 sides = [n.left, *n.comparators] 

160 for lhs, op, rhs in zip(sides, n.ops, sides[1:], strict=False): 

161 for me, other in ((lhs, rhs), (rhs, lhs)): 

162 if has_var(me) or (has_var(me, True) and use_of(other) == "categorical"): 

163 used.add(use_of(other, op)) 

164 elif isinstance(n, ast.Attribute) and has_var(n.value) and n.attr == "str": 

165 used.add("categorical") 

166 elif ( 

167 isinstance(n, ast.Call) 

168 and isinstance(n.func, ast.Attribute) 

169 and has_var(n.func.value) 

170 and n.func.attr in _CMP_METHODS 

171 ): 

172 args = [*n.args, *(k.value for k in n.keywords)] 

173 used.update(use_of(c) for a in args for c in ast.walk(a)) 

174 return used - {""}

177def _is_numeric_default(val_str: str) -> bool: 

178 s = val_str.strip() 

179 if len(s) >= 2 and s[0] in ("'", '"') and s[-1] == s[0]: 

180 return False 

181 return math.isfinite(_to_finite_float(s, float("nan"))) or s.lower() in ("nan", "+nan", "-nan")

185 "max": max, 

186 "min": min, 

187 "sum": sum, 

188 "mean": lambda x: float(sum(x) / len(x)), 

189 "first": lambda x: x[0], 

190 "last": lambda x: x[-1],

199 return default

202def _parse_default_float(val: Any) -> float: 

203 """Parse default float, preserving explicit nan strings/floats.""" 

204 is_nan = (isinstance(val, float) and np.isnan(val)) or ( 

205 isinstance(val, str) and val.strip().lower() in ("nan", "+nan", "-nan", ".nan") 

206 ) 

207 return float("nan") if is_nan else _to_finite_float(val, 0.0)

210def _parse_numeric_scalar( 

211 val: Any, 

212 default: float, 

213 reduction: str | None, 

214 col_name: str, 

215) -> float: 

216 """Parse a single value or delimited string into a numeric float.""" 

217 if val is None or pd.isna(val): 

218 return default 

219 if isinstance(val, (int, float, np.number)): 

220 return _to_finite_float(val, default) 

221 if isinstance(val, str): 

222 if (val := val.strip()) in _MISSING_TOKENS: 

223 return default 

224 if "," in val or ";" in val: 

225 tokens = [t.strip() for t in val.replace(";", ",").split(",")] 

226 valid_floats = [ 

227 f 

228 for t in tokens 

229 if t not in _MISSING_TOKENS 

230 if math.isfinite(f := _to_finite_float(t, float("nan"))) 

231 ] 

232 if not valid_floats: 

233 return default 

234 if len(valid_floats) == 1: 

235 return valid_floats[0] 

236 if not reduction: 

237 raise ValueError( 

238 f"Ambiguous multi-value numeric string '{val}' in column '{col_name}' " 

239 f"requires an explicit reduction policy ('max', 'min', etc.)." 

240 ) 

241 red = reduction.lower() 

242 if red in _REDUCTION_FUNCS: 

243 return _REDUCTION_FUNCS[red](valid_floats) 

244 raise ValueError(f"Unsupported reduction policy: '{reduction}'") 

245 return _to_finite_float(val, default)

272def _parse_var_config( 

273 original_col: str, var_config: Any 

274) -> tuple[str, str, str | None, str | None, bool] | None: 

275 if not isinstance(var_config, (str, dict)): 

276 return None 

277 target_name = str(original_col) 

278 default_val_str, reduction_policy, var_type, has_default = "0", None, None, False 

279 if isinstance(var_config, str): 

280 parts = var_config.split("|") 

281 target_name = parts[0].strip() or target_name 

282 for part in parts[1:]: 

283 if ":" in part: 

284 k, v = (x.strip() for x in part.split(":", 1)) 

285 if (kl := k.lower()) == "default": 

286 default_val_str, has_default = v, True 

287 elif kl in ("reduce", "reduction"): 

288 reduction_policy = v 

289 elif kl == "type": 

290 var_type = v.lower() 

291 else: 

292 target_name = str(var_config.get("target", original_col)) 

293 if "default" in var_config: 

294 default_val_str, has_default = str(var_config["default"]), True 

295 reduction_policy = var_config.get("reduce") or var_config.get("reduction") 

296 var_type = str(var_config.get("type", "")).lower() or None 

297 return target_name, default_val_str, reduction_policy, var_type, has_default

300def apply_scoring( 

301 df: pd.DataFrame, scoring_config: dict[str, Any], *, strict: bool = False 

302) -> pd.DataFrame: 

303 """Apply scoring formulas to variant DataFrame, evaluating and adding score columns.""" 

304 variables = scoring_config.get("variables", {}) 

305 formulas = scoring_config.get("formulas", []) 

306 

307 if not formulas: 

308 logger.debug("No scoring formulas provided. Skipping scoring.") 

309 return df 

310 

311 logger.info(f"Applying {len(formulas)} scoring formula(s)...") 

312 

313 scored_df = df.copy() 

314 calculated_columns: list[str] = [] 

315 parsed_vars: dict[str, tuple[str, str, str | None, str | None, bool]] = {} 

316 for col, cfg in variables.items(): 

317 parsed = _parse_var_config(str(col), cfg) 

318 if parsed: 

319 parsed_vars[str(col)] = parsed 

320 

321 # Fixed order: explicit type, bool dtype, known text column, default value type, numeric. 

322 resolved: dict[str, str] = {} 

323 for original_col, (target_name, default_str, _r, var_type, has_default) in parsed_vars.items(): 

324 col = scored_df.get(original_col) 

325 if var_type in ("bool", "boolean") or ( 

326 var_type is None and col is not None and pd.api.types.is_bool_dtype(col) 

327 ): 

328 resolved[target_name] = "bool" 

329 elif var_type in _NUMERIC_TYPES: 

330 resolved[target_name] = "numeric" 

331 elif ( 

332 var_type in _TEXT_TYPES 

333 or _is_known_categorical_column(original_col) 

334 or (has_default and not _is_numeric_default(default_str)) 

335 ): 

336 resolved[target_name] = "categorical" 

337 else: 

338 resolved[target_name] = "numeric" 

339 typed = { 

340 cfg[0]: resolved[cfg[0]] if cfg[3] or _is_known_categorical_column(col) else "" 

341 for col, cfg in parsed_vars.items() 

342 } 

343 # ponytail: transition guard (see _formula_usage) - raise where the old AST guesser's 

344 # precedence (string use > numeric use/reduce/numeric dtype > variable comparison) differs. 

345 for original_col, (target_name, _d, red, var_type, _h) in parsed_vars.items(): 

346 res, col = resolved[target_name], scored_df.get(original_col) 

347 used = _formula_usage(target_name, formulas, typed) 

348 numeric_dtype = col is not None and col.dtype.kind in "iuf" 

349 if red is not None or numeric_dtype or _numeric_col_name(original_col): 

350 used.add("numeric") 

351 old = next((u for u in ("categorical", "numeric") if u in used), min(used, default=res)) 

352 if var_type or _is_known_categorical_column(original_col) or old == res: 

353 continue 

354 if res != "bool" or old == "categorical": 

355 raise ValueError( 

356 f"Scoring variable '{target_name}' is ambiguous (formulas use it as {old}, " 

357 f"type rules resolve it to {res}): add '|type:numeric' or " 

358 "'|type:categorical' to its assignment in variable_assignment_config.json" 

359 ) 

360 

361 rename_map, created_vars = {}, [] 

362 for original_col, cfg_tuple in parsed_vars.items(): 

363 target_name, default_val_str, reduction_policy, var_type, has_default = cfg_tuple 

364 var_res = resolved[target_name] 

365 val_to_use = _clean_default_str(default_val_str) 

366 if original_col in scored_df.columns: 

367 if var_res == "bool": 

368 scored_df[original_col] = _convert_to_bool(scored_df[original_col], default_val_str) 

369 elif var_res == "numeric": 

370 s_col = scored_df[original_col] 

371 if not ( 

372 pd.api.types.is_integer_dtype(s_col) 

373 and not s_col.isna().any() 

374 and reduction_policy is None 

375 and var_type not in ("float",) 

376 ): 

377 scored_df[original_col] = convert_to_numeric( 

378 s_col, _parse_default_float(default_val_str), reduction_policy 

379 ) 

380 else: 

381 fill_str = val_to_use if has_default else "" 

382 s_orig = scored_df[original_col] 

383 s_cat = ( 

384 s_orig.astype(object) 

385 .map(lambda x, _f=fill_str: _f if pd.isna(x) else str(x)) 

386 .astype(str) 

387 ) 

388 scored_df[original_col] = s_cat.replace([".", "NA", ""], fill_str) 

389 rename_map[original_col] = target_name 

390 else: 

391 logger.warning( 

392 f"Variable column '{original_col}' not found. Using default '{default_val_str}'." 

393 ) 

394 scored_df[target_name] = ( 

395 pd.Series(_parse_default_bool(default_val_str), index=scored_df.index, dtype=bool) 

396 if var_res == "bool" 

397 else ( 

398 _parse_default_float(default_val_str) 

399 if var_res == "numeric" 

400 else (val_to_use if has_default else "") 

401 ) 

402 ) 

403 created_vars.append(target_name) 

404 

405 if rename_map: 

406 scored_df.rename(columns=rename_map, inplace=True) 

407 

408 for formula_item in formulas: 

409 for score_name, formula_str in formula_item.items(): 

410 logger.debug(f"Evaluating formula for '{score_name}': {formula_str}") 

411 try: 

412 scored_df[score_name] = scored_df.eval( 

413 formula_str, engine="python", local_dict={"np": np} 

414 ) 

415 if not pd.api.types.is_numeric_dtype(scored_df[score_name]): 

416 scored_df[score_name] = pd.to_numeric(scored_df[score_name], errors="coerce") 

417 logger.info(f"Successfully calculated score column: '{score_name}'") 

418 except Exception as e: 

419 logger.error(f"Error evaluating formula for '{score_name}': {e}") 

420 if strict: 

421 raise 

422 scored_df[score_name] = float("nan") 

423 calculated_columns.append(score_name) 

424 

425 if rename_map: 

426 scored_df.rename(columns={v: k for k, v in rename_map.items()}, inplace=True) 

427 if created_vars: 

428 scored_df.drop(columns=created_vars, inplace=True) 

429 

430 output_scores = scoring_config.get("output_scores") 

431 if output_scores is not None: 

432 columns_to_drop = [col for col in calculated_columns if col not in output_scores] 

433 if columns_to_drop: 

434 logger.info(f"Dropping intermediate columns: {columns_to_drop}") 

435 scored_df.drop(columns=columns_to_drop, inplace=True) 

436 

437 return scored_df
