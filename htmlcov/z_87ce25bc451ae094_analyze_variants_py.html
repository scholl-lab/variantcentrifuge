4""" 

5Variant analysis module for gene burden and other statistics. 

6 

7Modular design: 

8- Basic and comprehensive stats computations moved to stats.py. 

9- Gene burden analysis moved to gene_burden.py. 

10- Helper functions moved to helpers.py. 

11 

12This module now: 

13- Reads a TSV of variants and their annotations (including a GT column listing sample genotypes). 

14- Classifies samples into case/control sets based on user input (sample lists or phenotype terms). 

15- If no case/control criteria are provided, defaults to making all samples controls. 

16- Invokes external modules (stats.py, gene_burden.py) for computations. 

17- Orchestrates flow and writes or yields output lines. 

18 

19Maintains previous functionality, CLI interface, and output format. 

20"""

41def analyze_variants_df(df: pd.DataFrame, cfg: dict[str, Any]) -> pd.DataFrame: 

42 """ 

43 Analyze variants DataFrame and compute allele frequencies, counts, and statistics. 

44 

45 Parameters 

46 ---------- 

47 df : pd.DataFrame 

48 Input DataFrame with variant data. 

49 cfg : dict[str, Any] 

50 Configuration dictionary with sample lists, case/control assignments, stats settings. 

51 

52 Returns 

53 ------- 

54 pd.DataFrame 

55 Updated DataFrame with variant counts and statistics columns. 

56 """ 

57 if len(df) == 0: 

58 logger.debug("Empty DataFrame in analyze_variants_df.") 

59 return df.copy() 

60 

61 required_columns = ["CHROM", "POS", "REF", "ALT", "GENE", "GT"] 

62 missing_columns = [c for c in required_columns if c not in df.columns] 

63 if missing_columns: 

64 logger.error( 

65 "Missing required columns: %s. Returning unchanged DataFrame.", 

66 ", ".join(missing_columns), 

67 ) 

68 return df 

69 

70 if "sample_list" not in cfg or not cfg["sample_list"].strip(): 

71 logger.error("No sample_list found in cfg. Unable to determine the full sample set.") 

72 return df 

73 

74 df = df.copy(deep=False) 

75 all_samples = set(cfg["sample_list"].split(",")) 

76 logger.debug("Total samples from VCF header: %d", len(all_samples)) 

77 

78 case_samples, control_samples = determine_case_control_sets(all_samples, cfg, df) 

79 logger.debug( 

80 "Number of case samples: %d; Number of control samples: %d", 

81 len(case_samples), 

82 len(control_samples), 

83 ) 

84 

85 df = assign_case_control_counts(df, case_samples, control_samples, all_samples) 

86 

87 logger.debug("Computing basic statistics...") 

88 basic_stats_df = stats.compute_basic_stats(df, all_samples) 

89 logger.debug("Basic statistics computed.") 

90 

91 stats_output_file = cfg.get("stats_output_file") 

92 no_stats = cfg.get("no_stats", False) 

93 

94 if stats_output_file: 

95 logger.debug("Writing basic stats to %s", stats_output_file) 

96 use_compression = "gzip" if str(stats_output_file).endswith(".gz") else None 

97 if use_compression: 

98 basic_stats_df.to_csv( 

99 stats_output_file, sep="\t", index=False, header=True, mode="w", compression="gzip" 

100 ) 

101 else: 

102 basic_stats_df.to_csv(stats_output_file, sep="\t", index=False, header=True, mode="w") 

103 

104 if not no_stats: 

105 logger.debug("Computing comprehensive gene-level statistics...") 

106 comprehensive_df = stats.compute_gene_stats(df) 

107 impact_summary = stats.compute_impact_summary(df) 

108 variant_type_summary = stats.compute_variant_type_summary(df) 

109 

110 custom_stats = None 

111 stats_config_path = cfg.get("stats_config") 

112 if stats_config_path: 

113 logger.info(f"Computing custom statistics using config: {stats_config_path}") 

114 custom_stats = stats.compute_custom_stats(df, stats_config_path) 

115 

116 combined_stats = stats.merge_and_format_stats( 

117 comprehensive_df, impact_summary, variant_type_summary, custom_stats 

118 ) 

119 logger.debug("Comprehensive statistics computed.") 

120 

121 comp_stats_list = [] 

122 for _, row in combined_stats.iterrows(): 

123 gene = row["GENE"] 

124 for col in combined_stats.columns: 

125 if col == "GENE": 

126 continue 

127 val = row[col] 

128 comp_stats_list.append([f"{gene}_{col}", str(val)]) 

129 if comp_stats_list and stats_output_file: 

130 logger.debug("Appending comprehensive stats to the stats file.") 

131 comp_stats_df = pd.DataFrame(comp_stats_list, columns=["metric", "value"]) 

132 use_compression = "gzip" if str(stats_output_file).endswith(".gz") else None 

133 if use_compression: 

134 comp_stats_df.to_csv( 

135 stats_output_file, 

136 sep="\t", 

137 index=False, 

138 header=False, 

139 mode="a", 

140 compression="gzip", 

141 ) 

142 else: 

143 comp_stats_df.to_csv( 

144 stats_output_file, sep="\t", index=False, header=False, mode="a" 

145 ) 

146 

147 if custom_stats: 

148 if "dataset" in custom_stats and not custom_stats["dataset"].empty: 

149 logger.debug("Appending custom dataset stats to the stats file.") 

150 custom_stats["dataset"].to_csv( 

151 stats_output_file, sep="\t", index=False, header=False, mode="a" 

152 ) 

153 

154 if "groups" in custom_stats: 

155 for stat_name, stat_df in custom_stats["groups"].items(): 

156 logger.debug( 

157 f"Appending custom grouped stats '{stat_name}' to the stats file." 

158 ) 

159 for idx, row in stat_df.iterrows(): 

160 if isinstance(row, pd.Series): 

161 for col in stat_df.columns: 

162 if col not in stat_df.index.names: 

163 metric_name = f"{stat_name}_{idx}_{col}" 

164 comp_stats_list.append([metric_name, str(row[col])]) 

165 

166 if comp_stats_list: 

167 additional_stats_df = pd.DataFrame( 

168 comp_stats_list, columns=["metric", "value"] 

169 ) 

170 additional_stats_df.to_csv( 

171 stats_output_file, sep="\t", index=False, header=False, mode="a" 

172 ) 

173 else: 

174 logger.debug("No comprehensive stats requested (no_stats=True).") 

175 

176 return df

179def analyze_variants(lines: Iterator[str], cfg: dict[str, Any]) -> Iterator[str]: 

180 """ 

181 Analyze variants and optionally perform gene burden analysis. 

182 

183 Steps 

184 ----- 

185 1. Parse input TSV into a DataFrame. 

186 2. Compute counts and stats via analyze_variants_df. 

187 3. If requested, perform gene burden (Fisher's exact test + correction). 

188 

189 Parameters 

190 ---------- 

191 lines : Iterator[str] 

192 Input lines representing a TSV with variant data. 

193 cfg : Dict[str, Any] 

194 Configuration dictionary. 

195 

196 Yields 

197 ------ 

198 str 

199 Processed lines of output TSV or gene-level burden results. 

200 """ 

201 perform_gene_burden_flag = cfg.get("perform_gene_burden", False) 

202 

203 # Read all input lines into a text block 

204 text_data = "".join(line for line in lines) 

205 if not text_data.strip(): 

206 logger.debug("No input data provided to analyze_variants.") 

207 return 

208 

209 df = pd.read_csv(io.StringIO(text_data), sep="\t", dtype=str) 

210 if len(df) == 0: 

211 logger.debug("Empty DataFrame after reading input in analyze_variants.") 

212 header_line = text_data.strip().split("\n")[0] if text_data.strip() else "" 

213 if header_line: 

214 logger.debug("Returning only header row from analyze_variants.") 

215 yield header_line 

216 return 

217 

218 required_columns = ["CHROM", "POS", "REF", "ALT", "GENE", "GT"] 

219 missing_columns = [c for c in required_columns if c not in df.columns] 

220 if missing_columns: 

221 logger.error( 

222 "Missing required columns: %s. Returning unchanged lines.", 

223 ", ".join(missing_columns), 

224 ) 

225 for line in text_data.strip().split("\n"): 

226 yield line 

227 return 

228 

229 if "sample_list" not in cfg or not cfg["sample_list"].strip(): 

230 logger.error("No sample_list found in cfg. Unable to determine the full sample set.") 

231 for line in text_data.strip().split("\n"): 

232 yield line 

233 return 

234 

235 df = analyze_variants_df(df, cfg) 

236 

237 # Apply inheritance analysis if requested (for public analyze_variants callers) 

238 inheritance_already_done = ( 

239 "Inheritance_Pattern" in df.columns and "Inheritance_Details" in df.columns 

240 ) 

241 if ( 

242 cfg.get("calculate_inheritance") 

243 and cfg.get("pedigree_data") is not None 

244 and not inheritance_already_done 

245 and not cfg.get("skip_inheritance", False) 

246 ): 

247 logger.info("Performing inheritance pattern analysis...") 

248 sample_list = cfg.get("sample_list", "").split(",") if cfg.get("sample_list") else [] 

249 logger.debug(f"Sample list for inheritance: {sample_list}") 

250 use_vectorized = cfg.get("use_vectorized_comp_het", True) 

251 from .inheritance import analyze_inheritance 

252 

253 df = analyze_inheritance( 

254 df, 

255 cfg["pedigree_data"], 

256 sample_list, 

257 use_vectorized_comp_het=use_vectorized, 

258 reference=cfg.get("reference"), 

259 ) 

260 logger.info("Inheritance analysis complete.") 

261 elif inheritance_already_done: 

262 logger.info("Inheritance analysis already performed, skipping re-calculation.") 

263 

264 # Apply scoring if configuration is provided (can use inheritance results) 

265 scoring_config = cfg.get("scoring_config") 

266 if scoring_config: 

267 logger.info("Applying custom scoring model to variants.") 

268 from . import scoring 

269 

270 df = scoring.apply_scoring(df, scoring_config) 

271 logger.debug("Scoring applied successfully.") 

272 

273 # Perform gene burden analysis if requested 

274 if perform_gene_burden_flag: 

275 logger.debug("Performing configurable gene burden analysis...") 

276 gene_burden_results = gene_burden.perform_gene_burden_analysis(df, cfg) 

277 

278 if gene_burden_results.empty: 

279 logger.warning("Gene burden results are empty. Returning variant-level data.") 

280 out_str = df.to_csv(sep="\t", index=False) 

281 for line in out_str.strip().split("\n"): 

282 yield line 

283 return 

284 

285 burden_output_file = cfg.get("gene_burden_output_file", "gene_burden_results.tsv") 

286 logger.info("Writing gene burden results to %s", burden_output_file) 

287 # Apply compression to gene burden output 

288 use_compression = "gzip" if str(burden_output_file).endswith(".gz") else None 

289 gene_burden_results.to_csv( 

290 burden_output_file, sep="\t", index=False, compression=cast(Any, use_compression) 

291 ) 

292 

293 # Yield burden results as output 

294 burden_text = gene_burden_results.to_csv(sep="\t", index=False) 

295 for line in burden_text.strip().split("\n"): 

296 yield line 

297 else: 

298 logger.debug("Gene burden not requested, returning processed data.") 

299 out_str = df.to_csv(sep="\t", index=False) 

300 for line in out_str.strip().split("\n"): 

301 yield line
