2""" 

3Heterogeneity diagnostics for cohort meta-analysis. 

4 

5- Per-stratum burden effect and standard error from the score/covariance. 

6- Cochran's Q, its p-value, and I-squared across strata with non-zero information 

7 (df = usable strata - 1). 

8"""

22def per_stratum_burden( 

23 u_list: list[np.ndarray], 

24 v_list: list[np.ndarray], 

25 weights: np.ndarray, 

26) -> tuple[np.ndarray, np.ndarray]: 

27 """Return per-stratum burden effect estimates and standard errors. 

28 

29 ``beta_c = (wᵀU_c) / (wᵀV_c w)``, ``se_c = 1/sqrt(wᵀV_c w)``. Strata whose 

30 information is numerically zero get ``beta=nan``, ``se=inf`` (same rule as 

31 ``combine.meta_burden``: below _DECISION_ACC of its Cauchy-Schwarz scale). 

32 """ 

33 w = np.asarray(weights, dtype=np.float64) 

34 betas: list[float] = [] 

35 ses: list[float] = [] 

36 for u, v in zip(u_list, v_list, strict=True): 

37 score = float(w @ np.asarray(u, dtype=np.float64)) 

38 v = np.asarray(v, dtype=np.float64) 

39 info = float(w @ (v @ w)) 

40 scale = float((np.abs(w) * np.sqrt(np.clip(np.diag(v), 0.0, None))).sum()) ** 2 

41 if info <= _DECISION_ACC * scale: 

42 betas.append(np.nan) 

43 ses.append(np.inf) 

44 else: 

45 betas.append(score / info) 

46 ses.append(1.0 / np.sqrt(info)) 

47 return np.array(betas), np.array(ses)

50def cochran_q(betas: np.ndarray, ses: np.ndarray) -> tuple[float, float, float]: 

51 """Cochran's Q heterogeneity test. Returns (Q, p_value, I²). 

52 

53 Uses only strata with finite beta and positive, finite SE. df = usable - 1. 

54 """ 

55 betas = np.asarray(betas, dtype=np.float64) 

56 ses = np.asarray(ses, dtype=np.float64) 

57 usable = np.isfinite(betas) & np.isfinite(ses) & (ses > 0) 

58 b = betas[usable] 

59 s = ses[usable] 

60 k = b.size 

61 if k < 2: 

62 return 0.0, 1.0, 0.0 

63 w = 1.0 / (s**2) 

64 beta_bar = float(np.sum(w * b) / np.sum(w)) 

65 q = float(np.sum(w * (b - beta_bar) ** 2)) 

66 df = k - 1 

67 p = float(scipy.stats.chi2.sf(q, df)) 

68 i2 = max(0.0, (q - df) / q) if q > 0 else 0.0 

69 return q, p, i2
