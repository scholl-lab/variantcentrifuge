80class AssociationEngine: 

81 """ 

82 Orchestrates association testing across multiple genes and multiple tests. 

83 

84 Usage 

85 ----- 

86 >>> config = AssociationConfig(gene_burden_mode="samples") 

87 >>> engine = AssociationEngine.from_names(["fisher"], config) 

88 >>> result_df = engine.run_all(gene_burden_data) 

89 

90 Parameters 

91 ---------- 

92 tests : list of AssociationTest 

93 Instantiated test objects to run. Use from_names() for the common case. 

94 config : AssociationConfig 

95 Configuration shared across all tests. 

96 """ 

97 

98 unit_label = "genes" # "gene sets" on the set engine; labels progress and summary lines 

99 

100 def __init__(self, tests: list[AssociationTest], config: AssociationConfig) -> None: 

101 self._tests: dict[str, AssociationTest] = {t.name: t for t in tests} 

102 self._config = config 

103 self._primary_override = resolve_primary_test(list(self._tests), config) 

104 

105 @classmethod 

106 def from_names( 

107 cls, 

108 test_names: list[str], 

109 config: AssociationConfig, 

110 ) -> AssociationEngine: 

111 """ 

112 Construct engine from a list of test name strings. 

113 

114 Parameters 

115 ---------- 

116 test_names : list of str 

117 Names of tests to run (e.g. ["fisher"]). 

118 config : AssociationConfig 

119 Runtime configuration. 

120 

121 Returns 

122 ------- 

123 AssociationEngine 

124 

125 Raises 

126 ------ 

127 ValueError 

128 If any test_name is not in the registry. The error message lists 

129 all available tests so users know what's valid. Note: "acat_o" is 

130 NOT a valid test name here — ACAT-O is computed post-loop as a 

131 meta-test and does not run per-gene. 

132 ImportError 

133 If a test's required dependencies are not installed. Raised eagerly 

134 (before any data processing) by check_dependencies(). 

135 """ 

136 registry = _build_registry() 

137 

138 # Backend-aware swap: --skat-backend python/auto routes "skat" to PurePythonSKATTest. 

139 # This runs BEFORE the unknown-name check so that "skat" resolves correctly. 

140 # "auto" and "python" follow identical code paths (unified pattern). 

141 skat_backend = getattr(config, "skat_backend", "python") 

142 if skat_backend in ("python", "auto"): 

143 from variantcentrifuge.association.tests.skat_python import PurePythonSKATTest 

144 

145 registry["skat"] = PurePythonSKATTest 

146 

147 # Backend-aware swap: --coast-backend python/auto routes "coast" to PurePythonCOASTTest. 

148 # This runs BEFORE the unknown-name check so that "coast" resolves correctly. 

149 # "auto" and "python" follow identical code paths (unified pattern). 

150 coast_backend = getattr(config, "coast_backend", "python") 

151 if coast_backend in ("python", "auto"): 

152 from variantcentrifuge.association.tests.allelic_series_python import ( 

153 PurePythonCOASTTest, 

154 ) 

155 

156 registry["coast"] = PurePythonCOASTTest 

157 

158 available = sorted(registry.keys()) 

159 

160 unknown = [name for name in test_names if name not in registry] 

161 if unknown: 

162 for name in unknown: 

163 raise ValueError( 

164 f"Test '{name}' is not available. Available tests: {', '.join(available)}" 

165 ) 

166 

167 if skat_backend == "r" and "skat" in test_names: 

168 from variantcentrifuge.association.tests.skat_r import _is_r_skat_weight_supported 

169 

170 if not _is_r_skat_weight_supported(config.variant_weights): 

171 raise ValueError( 

172 "R SKAT backend supports only beta:a,b and uniform variant weights. " 

173 f"Got '{config.variant_weights}'. Use --skat-backend python for cadd, revel, " 

174 "combined, score_column, or column:<name> weights." 

175 ) 

176 

177 tests: list[AssociationTest] = [] 

178 for name in test_names: 

179 test_cls = registry[name] 

180 test = test_cls() 

181 test.check_dependencies() # eager dependency check 

182 tests.append(test) 

183 

184 return cls(tests, config)

195 def run_all( 

196 self, gene_burden_data: list[dict[str, Any]], gene_hook: GeneHook | None = None 

197 ) -> pd.DataFrame: 

198 """ 

199 Run all registered tests across all genes and return wide-format results. 

200 

201 FDR correction is applied once to the primary p-value across all genes. 

202 Multi-test and SKAT runs use ACAT-O as the primary result. Single 

203 non-SKAT runs use the test's own p-value as the primary result. 

204 

205 Parameters 

206 ---------- 

207 gene_burden_data : list of dict 

208 One dict per gene with keys: GENE, proband_count, control_count, 

209 proband_carrier_count, control_carrier_count, proband_allele_count, 

210 control_allele_count, n_qualifying_variants. 

211 gene_hook : callable, optional 

212 ``gene_hook(gene, builder, build_result)`` runs (in the worker) right after 

213 a gene's matrix is built; return values are in ``self.gene_hook_results``. 

214 

215 Returns 

216 ------- 

217 pd.DataFrame 

218 Wide-format results. One row per gene that has at least one test 

219 result (genes where all tests returned p_value=None are excluded). 

220 Columns: gene, n_cases, n_controls, n_variants, then per-test 

221 columns: {test}_pvalue, plus test-aware effect columns (e.g. 

222 fisher_or or logistic_burden_beta), and primary_pvalue/primary_qvalue. 

223 ACAT-O columns are present only when ACAT-O is the primary result. 

224 """ 

225 self.gene_hook_results: dict[str, Any] = {} 

226 self.run_info: list[str] = [] 

227 label = self.unit_label 

228 if not gene_burden_data: 

229 logger.warning("No gene burden data provided to AssociationEngine.") 

230 return pd.DataFrame() 

231 

232 # Sort by gene name for deterministic correction order (matches gene_burden.py line 411) 

233 sorted_data = sorted(gene_burden_data, key=lambda d: d.get("GENE", "")) 

234 

235 logger.info( 

236 f"Association analysis: running {list(self._tests.keys())} " 

237 f"tests on {len(sorted_data)} genes" 

238 ) 

239 

240 # Collect results: results_by_test[test_name][gene] = TestResult 

241 results_by_test: dict[str, dict[str, TestResult]] = {name: {} for name in self._tests} 

242 

243 # Lifecycle hook: prepare() before gene loop (allows progress setup, panel warnings) 

244 for test in self._tests.values(): 

245 test.prepare(len(sorted_data)) 

246 

247 # Determine whether to use parallel execution (Phase 27) 

248 n_workers = self._config.association_workers 

249 all_parallel_safe = all( 

250 getattr(test, "parallel_safe", False) for test in self._tests.values() 

251 ) 

252 use_parallel = ( 

253 n_workers != 1 and all_parallel_safe and len(sorted_data) >= MIN_GENES_FOR_PARALLEL 

254 ) 

255 

256 if n_workers != 1 and not all_parallel_safe: 

257 logger.warning( 

258 "association_workers=%d requested but not all tests are parallel_safe " 

259 "(falling back to sequential). Non-parallel tests: %s", 

260 n_workers, 

261 [n for n, t in self._tests.items() if not getattr(t, "parallel_safe", False)], 

262 ) 

263 

264 if use_parallel: 

265 workers_note = f"association_workers={n_workers}, parallel" 

266 elif n_workers == 1: 

267 workers_note = "association_workers=1, sequential" 

268 elif not all_parallel_safe: 

269 workers_note = f"association_workers={n_workers} but tests not parallel_safe" 

270 else: 

271 workers_note = f"association_workers={n_workers} but < {MIN_GENES_FOR_PARALLEL} genes" 

272 self.run_info = [f"gene loop: {workers_note}; BLAS pinned to 1 thread"] 

273 

274 with single_blas_thread(): # gene-level parallelism only, never BLAS (#212) 

275 if use_parallel: 

276 # Run genes in the parent until every lazy null model is fitted (or failed), so 

277 # forked workers inherit it instead of each refitting it (#212). 

278 done = 0 

279 while done < len(sorted_data) and ( 

280 done == 0 

281 or (done < MIN_GENES_FOR_PARALLEL and _null_model_pending(self._tests)) 

282 ): 

283 run_sequential_genes( 

284 sorted_data[done : done + 1], 

285 self._tests, 

286 self._config, 

287 results_by_test, 

288 gene_hook, 

289 self.gene_hook_results, 

290 ) 

291 done += 1 

292 

293 # Dispatch remaining genes to worker processes 

294 remaining = sorted_data[done:] 

295 if remaining: 

296 actual_workers = (os.cpu_count() or 1) if n_workers == -1 else n_workers 

297 # Don't over-provision workers for small remaining panels 

298 if len(remaining) < actual_workers * 2: 

299 actual_workers = max(1, len(remaining) // 2) 

300 

301 dispatch_parallel_genes( 

302 remaining=remaining, 

303 tests=self._tests, 

304 config=self._config, 

305 actual_workers=actual_workers, 

306 results_by_test=results_by_test, 

307 gene_hook=gene_hook, 

308 hook_results=self.gene_hook_results, 

309 progress=Progress(f"Association {label}", len(sorted_data), offset=done), 

310 ) 

311 else: 

312 # Sequential gene loop (default path or fallback) 

313 run_sequential_genes( 

314 sorted_data, 

315 self._tests, 

316 self._config, 

317 results_by_test, 

318 gene_hook, 

319 self.gene_hook_results, 

320 Progress(f"Association {self.unit_label}", len(sorted_data)), 

321 ) 

322 

323 for test_name, by_gene in results_by_test.items(): 

324 skips = Counter( 

325 str(code) 

326 for r in by_gene.values() 

327 if (code := r.extra.get("skat_skip_reason") or r.extra.get("coast_skip_reason")) 

328 ) 

329 if skips: 

330 self.run_info.append(f"{label} skipped per test - {test_name}: {dict(skips)}") 

331 

332 # Lifecycle hook: finalize() after gene loop (allows timing summary, cleanup) 

333 for test in self._tests.values(): 

334 test.finalize() 

335 

336 if not self._tests: 

337 logger.warning("Association analysis: no association tests registered.") 

338 return pd.DataFrame() 

339 

340 # Single correction pass on the run's primary p-value. 

341 # 

342 # ACAT-O is the right primary result for multi-test runs. It also stays 

343 # primary for SKAT because the engine can include ACAT-V in that omnibus 

344 # combination. For single non-SKAT tests, ACAT-O would be a pure 

345 # pass-through, so correction is attached to the actual test result 

346 # (for example fisher_qvalue) instead of misleading acat_o_* columns. 

347 # --primary-test skato/burden overrides the primary; ACAT-O is then still 

348 # reported (acat_o_pvalue) but not corrected (#210). 

349 include_acat_v = getattr(self._config, "acat_o_exclude", None) != "acat_v" 

350 write_acat_o = len(self._tests) > 1 or ("skat" in self._tests and include_acat_v) 

351 acat_o_results: dict[str, TestResult] = {} 

352 if write_acat_o: 

353 acat_o_results = self._compute_acat_o(results_by_test, sorted_data) 

354 results_by_test["acat_o"] = acat_o_results 

355 primary_result_id = self._primary_override or ( 

356 "acat_o" if write_acat_o else next(iter(self._tests)) 

357 ) 

358 acat_o_primary = primary_result_id == "acat_o" 

359 correction_results: dict[str, TestResult] = results_by_test[primary_result_id] 

360 

361 gene_order = [d.get("GENE", "") for d in sorted_data] 

362 testable_genes = [ 

363 g 

364 for g in gene_order 

365 if correction_results.get(g) is not None and correction_results[g].p_value is not None 

366 ] 

367 

368 self.run_info.insert( 

369 1, 

370 f"{label}: {len(sorted_data)} given, {len(testable_genes)} with a primary result, " 

371 f"{len(sorted_data) - len(testable_genes)} without", 

372 ) 

373 

374 # Phase 33: weighted or unweighted correction path 

375 fdr_weights_by_gene: dict[str, float] = {} 

376 unweighted_corrected_by_gene: dict[str, float] = {} 

377 weighted_corrected_by_gene: dict[str, float] = {} 

378 weighted_norm_arr: list[float] = [] 

379 

380 if testable_genes: 

381 raw_pvals: list[float] = [ 

382 correction_results[g].p_value # type: ignore[misc] 

383 for g in testable_genes 

384 ] 

385 

386 use_weighted = bool(getattr(self._config, "gene_prior_weights", None)) 

387 

388 if use_weighted: 

389 from variantcentrifuge.association.correction import ( 

390 apply_weighted_correction, 

391 load_gene_weights, 

392 ) 

393 

394 weight_col = getattr(self._config, "gene_prior_weight_column", "weight") 

395 weight_map = load_gene_weights( 

396 self._config.gene_prior_weights, # type: ignore[arg-type] 

397 weight_col, 

398 ) 

399 

400 # Unweighted for diagnostics comparison 

401 unweighted = apply_correction(raw_pvals, self._config.correction_method) 

402 for g, uw in zip(testable_genes, unweighted, strict=True): 

403 unweighted_corrected_by_gene[g] = float(uw) 

404 

405 corrected, norm_w = apply_weighted_correction( 

406 raw_pvals, testable_genes, weight_map, self._config.correction_method 

407 ) 

408 weighted_norm_arr = norm_w.tolist() 

409 for g, nw in zip(testable_genes, norm_w, strict=True): 

410 fdr_weights_by_gene[g] = float(nw) 

411 for g, wt in zip(testable_genes, corrected, strict=True): 

412 weighted_corrected_by_gene[g] = float(wt) 

413 

414 # Write diagnostics if requested (Phase 33) 

415 diag_out = getattr(self._config, "diagnostics_output", None) 

416 if diag_out and len(testable_genes) > 1: 

417 import numpy as _np 

418 

419 from variantcentrifuge.association.diagnostics import ( 

420 write_fdr_weight_diagnostics, 

421 ) 

422 

423 write_fdr_weight_diagnostics( 

424 genes=testable_genes, 

425 raw_weights=weight_map, 

426 normalized_weights=_np.asarray(weighted_norm_arr, dtype=float), 

427 unweighted_corrected=_np.asarray( 

428 [unweighted_corrected_by_gene[g] for g in testable_genes], 

429 dtype=float, 

430 ), 

431 weighted_corrected=_np.asarray( 

432 [float(corrected[i]) for i in range(len(testable_genes))], 

433 dtype=float, 

434 ), 

435 fdr_threshold=0.05, 

436 diagnostics_dir=diag_out, 

437 ) 

438 else: 

439 corrected = apply_correction(raw_pvals, self._config.correction_method) 

440 

441 for gene, corr_p in zip(testable_genes, corrected, strict=True): 

442 correction_results[gene].corrected_p_value = float(corr_p) 

443 

444 # Build wide-format DataFrame 

445 rows = [] 

446 for gene_data in sorted_data: 

447 gene = gene_data.get("GENE", "") 

448 # Include gene if it has a result, a skip reason or a warning code 

449 any_result = any(results_by_test[tn][gene].p_value is not None for tn in self._tests) 

450 any_skip = any( 

451 (extra := results_by_test[tn][gene].extra).get("skat_skip_reason") 

452 or extra.get("coast_skip_reason") 

453 or extra.get("warnings") 

454 for tn in self._tests 

455 ) 

456 if not any_result and not any_skip: 

457 continue 

458 

459 # Use metadata from first result with data, or first for skipped 

460 first_result = next( 

461 (r for tn in self._tests if (r := results_by_test[tn][gene]).p_value is not None), 

462 next(results_by_test[tn][gene] for tn in self._tests), 

463 ) 

464 

465 row: dict[str, Any] = { 

466 "gene": gene, 

467 "n_cases": first_result.n_cases, 

468 "n_controls": first_result.n_controls, 

469 "n_variants": first_result.n_variants, 

470 } 

471 

472 for test_name, test in self._tests.items(): 

473 res = results_by_test[test_name][gene] 

474 col_names = test.effect_column_names() 

475 row[f"{test_name}_pvalue"] = res.p_value 

476 # Corrected significance is emitted through primary_* columns 

477 # and, for single-test runs, through {test}_qvalue aliases. 

478 # Do not reintroduce legacy {test}_corrected_pvalue columns. 

479 # None-effect guard: skip column creation when effect/CI names are None. 

480 # Required for SKAT which has no effect size, SE, or confidence interval. 

481 # Without this guard, col_names['effect'] = None would produce 

482 # "skat_None" as a column name. 

483 if col_names.get("effect") is not None: 

484 row[f"{test_name}_{col_names['effect']}"] = res.effect_size 

485 if col_names.get("se") is not None: 

486 row[f"{test_name}_{col_names['se']}"] = res.se 

487 if col_names.get("ci_lower") is not None: 

488 row[f"{test_name}_{col_names['ci_lower']}"] = res.ci_lower 

489 if col_names.get("ci_upper") is not None: 

490 row[f"{test_name}_{col_names['ci_upper']}"] = res.ci_upper 

491 # Write test-specific extra columns (e.g. skat_o_rho, skat_warnings). 

492 # Keys in res.extra are already namespaced by the test (e.g. "skat_o_rho"), 

493 # so we write them without an additional test_name prefix. 

494 # Fisher and burden tests have empty extra dicts, so this is a no-op for them. 

495 for extra_key, extra_val in res.extra.items(): 

496 row[extra_key] = extra_val 

497 

498 # SKAT-O post-processing: rename skat_pvalue -> skat_o_pvalue when 

499 # skat_method is SKATO. The skat_method key is stored in res.extra 

500 # and already written to the row above. 

501 if res.extra.get("skat_method") == "SKATO" and f"{test_name}_pvalue" in row: 

502 row[f"{test_name}_o_pvalue"] = row.pop(f"{test_name}_pvalue") 

503 

504 # Primary significance columns. In single-test pass-through mode, 

505 # also expose a test-specific q-value such as fisher_qvalue. 

506 primary_res = correction_results.get(gene) 

507 primary_output_id = primary_result_id 

508 if not acat_o_primary: 

509 primary_output_id = ( 

510 "skat_o" 

511 if self._primary_override == "skat" 

512 else _single_test_output_id(primary_result_id, primary_res) 

513 ) 

514 row[f"{primary_output_id}_qvalue"] = ( 

515 primary_res.corrected_p_value if primary_res is not None else None 

516 ) 

517 if weighted_corrected_by_gene: 

518 row[f"{primary_output_id}_weighted_qvalue"] = weighted_corrected_by_gene.get( 

519 gene 

520 ) 

521 row[f"{primary_output_id}_unweighted_qvalue"] = ( 

522 unweighted_corrected_by_gene.get(gene) 

523 ) 

524 

525 row["primary_test"] = primary_output_id 

526 row["primary_pvalue"] = primary_res.p_value if primary_res is not None else None 

527 row["primary_qvalue"] = ( 

528 primary_res.corrected_p_value if primary_res is not None else None 

529 ) 

530 if weighted_corrected_by_gene: 

531 row["primary_weighted_qvalue"] = weighted_corrected_by_gene.get(gene) 

532 row["primary_unweighted_qvalue"] = unweighted_corrected_by_gene.get(gene) 

533 

534 # ACAT-O columns are only emitted when ACAT-O is a real combination, 

535 # not for single-test pass-throughs such as Fisher-only; its q-values 

536 # only when it is the primary (the corrected family). 

537 if write_acat_o: 

538 acat_res = acat_o_results.get(gene) 

539 row["acat_o_pvalue"] = acat_res.p_value if acat_res is not None else None 

540 if acat_o_primary: 

541 row["acat_o_qvalue"] = ( 

542 acat_res.corrected_p_value if acat_res is not None else None 

543 ) 

544 if weighted_corrected_by_gene: 

545 row["acat_o_weighted_qvalue"] = weighted_corrected_by_gene.get(gene) 

546 row["acat_o_unweighted_qvalue"] = unweighted_corrected_by_gene.get(gene) 

547 

548 # Phase 33: fdr_weight column — only added when weighted BH is active 

549 if fdr_weights_by_gene: 

550 row["fdr_weight"] = fdr_weights_by_gene.get(gene, 1.0) 

551 

552 rows.append(row) 

553 

554 if not rows: 

555 logger.warning("Association analysis: no genes with testable variants.") 

556 return pd.DataFrame() 

557 

558 result_df = pd.DataFrame(rows) 

559 n_sig = (result_df.get("primary_qvalue", pd.Series(dtype=float)) < 0.05).sum() 

560 logger.info( 

561 f"Association analysis complete: {len(result_df)} genes tested, " 

562 f"{n_sig} significant (primary corrected p < 0.05)"
