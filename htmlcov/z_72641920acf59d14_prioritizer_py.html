43def prioritize_patterns( 

44 patterns: list[str], 

45 segregation_results: dict[str, tuple[bool, float]] | None = None, 

46) -> tuple[str, float]: 

47 """ 

48 Prioritize inheritance patterns and return the highest priority pattern. 

49 

50 Parameters 

51 ---------- 

52 patterns : List[str] 

53 List of possible inheritance patterns 

54 segregation_results : Optional[Dict[str, Tuple[bool, float]]] 

55 Optional segregation check results 

56 

57 Returns 

58 ------- 

59 Tuple[str, float] 

60 Tuple of (highest_priority_pattern, confidence_score) 

61 """ 

62 if not patterns: 

63 return ("unknown", 0.0) 

64 

65 # Filter patterns by segregation if results available 

66 if segregation_results: 

67 # Separate patterns that segregate vs those that don't 

68 segregating_patterns = [] 

69 non_segregating_patterns = [] 

70 

71 for pattern in patterns: 

72 if pattern in segregation_results: 

73 segregates, seg_confidence = segregation_results[pattern] 

74 if segregates and seg_confidence > 0.5: 

75 segregating_patterns.append(pattern) 

76 else: 

77 non_segregating_patterns.append(pattern) 

78 else: 

79 # If no segregation data, include in segregating list 

80 segregating_patterns.append(pattern) 

81 

82 # Use segregating patterns if any exist, otherwise fall back to non-segregating 

83 if segregating_patterns: 

84 patterns = segregating_patterns 

85 elif non_segregating_patterns: 

86 patterns = non_segregating_patterns 

87 

88 # Calculate scores for each pattern 

89 pattern_scores = {} 

90 for pattern in patterns: 

91 base_score = PATTERN_PRIORITY.get(pattern, 0) 

92 pattern_scores[pattern] = base_score 

93 

94 # Get highest scoring pattern 

95 best_pattern = max(pattern_scores.items(), key=lambda x: x[1]) 

96 

97 # Calculate confidence based on score distribution 

98 confidence = calculate_confidence(pattern_scores, best_pattern[0]) 

99 

100 return (best_pattern[0], confidence)

103def calculate_confidence(pattern_scores: Mapping[str, int | float], best_pattern: str) -> float: 

104 """ 

105 Calculate confidence score based on pattern score distribution. 

106 

107 Parameters 

108 ---------- 

109 pattern_scores : Dict[str, float] 

110 Dictionary of pattern scores 

111 best_pattern : str 

112 The highest scoring pattern 

113 

114 Returns 

115 ------- 

116 float 

117 Confidence score between 0 and 1 

118 """ 

119 if not pattern_scores: 

120 return 0.0 

121 

122 best_score = pattern_scores[best_pattern] 

123 

124 # If only one pattern, confidence depends on its priority 

125 if len(pattern_scores) == 1: 

126 return min(best_score / 100.0, 1.0) 

127 

128 # Calculate score separation 

129 other_scores = [score for pattern, score in pattern_scores.items() if pattern != best_pattern] 

130 

131 if not other_scores: 

132 return 1.0 

133 

134 second_best = max(other_scores) 

135 score_separation = best_score - second_best 

136 

137 # Confidence based on separation and absolute score 

138 confidence = min((score_separation / 50.0) * (best_score / 100.0), 1.0) 

139 

140 return round(confidence, 2)
