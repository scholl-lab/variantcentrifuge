1"""ClinVar PM5 annotation — runtime module. 

2 

3Loads a pre-built PM5 lookup table (from ``variantcentrifuge-build-pm5``) 

4and annotates variants with ACMG PM5 criterion information. 

5 

6PM5: "Novel missense change at an amino acid residue where a different 

7missense change determined to be pathogenic has been seen before." 

8 

9Three strength levels: 

10- PM5_Strong — 2+ independent Pathogenic variants at same residue 

11- PM5 — 1 Pathogenic variant at same residue 

12- PM5_Supporting — only Likely_pathogenic evidence 

13""" 

14 

15from __future__ import annotations 

16 

17import gzip 

18import json 

19import logging 

20import re 

21from typing import Any 

22 

23import pandas as pd 

24 

25logger = logging.getLogger("variantcentrifuge")

27# --------------------------------------------------------------------------- 

28# Amino acid code tables 

29# --------------------------------------------------------------------------- 

30AA_3TO1: dict[str, str] = { 

31 "Ala": "A", 

32 "Arg": "R", 

33 "Asn": "N", 

34 "Asp": "D", 

35 "Cys": "C", 

36 "Gln": "Q", 

37 "Glu": "E", 

38 "Gly": "G", 

39 "His": "H", 

40 "Ile": "I", 

41 "Leu": "L", 

42 "Lys": "K", 

43 "Met": "M", 

44 "Phe": "F", 

45 "Pro": "P", 

46 "Ser": "S", 

47 "Thr": "T", 

48 "Trp": "W", 

49 "Tyr": "Y", 

50 "Val": "V", 

51 "Ter": "*", 

52} 

53AA_1TO3: dict[str, str] = {v: k for k, v in AA_3TO1.items()} 

54 

55# --------------------------------------------------------------------------- 

56# HGVS protein notation parsers 

57# --------------------------------------------------------------------------- 

58# 3-letter: p.Arg123His, p.(Arg123His), p.Arg123* 

59# The trailing (?![a-z]) prevents matching frameshifts like p.Arg123Profs*45 

60HGVS_P_3LETTER = re.compile(r"p\.\(?([A-Z][a-z]{2})(\d+)([A-Z][a-z]{2}|\*)(?![a-z])\)?") 

61# 1-letter: p.R123H, p.(R123H), p.R123* 

62HGVS_P_1LETTER = re.compile(r"p\.\(?([A-Z])(\d+)([A-Z]|\*)(?![a-z])\)?") 

63 

64NOT_TRIGGERED: dict[str, Any] = { 

65 "triggered": False, 

66 "strength": None, 

67 "evidence_count": 0, 

68 "known_variants": "", 

69}

72def normalize_aa(aa: str) -> str: 

73 """Normalize amino acid to 1-letter code for comparison.""" 

74 return AA_3TO1.get(aa, aa)

77def parse_hgvs_p(hgvs_p: str) -> tuple[str, int, str] | None: 

78 """Extract (ref_aa, position, alt_aa) from HGVS protein notation. 

79 

80 Returns None for unparseable notation (p.?, p.=, frameshifts, etc.). 

81 """ 

82 if not hgvs_p or hgvs_p in ("NA", ".", "p.?", "p.="): 

83 return None 

84 for pattern in (HGVS_P_3LETTER, HGVS_P_1LETTER): 

85 m = pattern.search(str(hgvs_p)) 

86 if m: 

87 return m.group(1), int(m.group(2)), m.group(3) 

88 return None

91def check_pm5( 

92 gene: str, 

93 ref_aa: str, 

94 aa_pos: int, 

95 alt_aa: str, 

96 lookup: dict[str, list[dict[str, str]]], 

97) -> dict[str, Any]: 

98 """Check ACMG PM5 criterion for a variant. 

99 

100 Parameters 

101 ---------- 

102 gene : str 

103 Gene symbol. 

104 ref_aa : str 

105 Reference amino acid (1- or 3-letter). 

106 aa_pos : int 

107 Amino acid position. 

108 alt_aa : str 

109 Alternate amino acid (1- or 3-letter). 

110 lookup : dict 

111 PM5 lookup table: ``{"GENE:pos": [{"ref_aa", "alt_aa", "clnsig", "review"}]}``. 

112 

113 Returns 

114 ------- 

115 dict 

116 Keys: triggered, strength, evidence_count, known_variants. 

117 """ 

118 key = f"{gene}:{aa_pos}" 

119 entries = lookup.get(key, []) 

120 if not entries: 

121 return NOT_TRIGGERED.copy() 

122 

123 query_alt = normalize_aa(alt_aa) 

124 query_ref = normalize_aa(ref_aa) 

125 

126 evidence: list[dict[str, str]] = [] 

127 for entry in entries: 

128 entry_ref = normalize_aa(entry["ref_aa"]) 

129 entry_alt = normalize_aa(entry["alt_aa"]) 

130 

131 # Reference AA must match 

132 if entry_ref != query_ref: 

133 continue 

134 

135 # Same AA change = PS1, NOT PM5 — skip 

136 if entry_alt == query_alt: 

137 continue 

138 

139 evidence.append(entry) 

140 

141 if not evidence: 

142 return NOT_TRIGGERED.copy() 

143 

144 p_count = sum(1 for e in evidence if e["clnsig"] == "P") 

145 

146 if p_count >= 2: 

147 strength = "PM5_Strong" 

148 elif p_count >= 1: 

149 strength = "PM5" 

150 else: 

151 strength = "PM5_Supporting" 

152 

153 known = ",".join(f"{e['ref_aa']}{aa_pos}{e['alt_aa']}({e['clnsig']})" for e in evidence) 

154 

155 return { 

156 "triggered": True, 

157 "strength": strength, 

158 "evidence_count": len(evidence), 

159 "known_variants": known, 

160 }

163def load_pm5_lookup(path: str) -> dict[str, list[dict[str, str]]]: 

164 """Load a PM5 lookup table from JSON or gzipped JSON. 

165 

166 Parameters 

167 ---------- 

168 path : str 

169 Path to the lookup file (.json or .json.gz). 

170 

171 Returns 

172 ------- 

173 dict 

174 The ``"lookup"`` section of the file. 

175 """ 

176 opener = gzip.open if path.endswith(".gz") else open 

177 with opener(path, "rt", encoding="utf-8") as f: 

178 data = json.load(f) 

179 

180 meta = data.get("_meta", {}) 

181 logger.info( 

182 "Loaded PM5 lookup: %d positions, %d entries, assembly=%s, built=%s", 

183 meta.get("position_count", "?"), 

184 meta.get("entry_count", "?"), 

185 meta.get("assembly", "?"), 

186 meta.get("build_date", "?"), 

187 ) 

188 result: dict[str, list[dict[str, str]]] = data.get("lookup", {}) 

189 return result

192def annotate_pm5( 

193 df: pd.DataFrame, 

194 lookup: dict[str, list[dict[str, str]]], 

195 gene_col: str = "GENE", 

196 hgvs_col: str = "HGVS_P", 

197) -> pd.DataFrame: 

198 """Add PM5 annotation columns to a variant DataFrame. 

199 

200 Adds three columns: ``PM5``, ``PM5_evidence_count``, ``PM5_known_variants``. 

201 

202 Parameters 

203 ---------- 

204 df : pd.DataFrame 

205 Variant table. 

206 lookup : dict 

207 PM5 lookup from :func:`load_pm5_lookup`. 

208 gene_col : str 

209 Column name for gene symbol. 

210 hgvs_col : str 

211 Column name for HGVS protein notation. 

212 

213 Returns 

214 ------- 

215 pd.DataFrame 

216 The input DataFrame with PM5 columns added. 

217 """ 

218 results: list[dict[str, Any]] = [] 

219 

220 genes = df[gene_col].astype(str).tolist() if gene_col in df.columns else [""] * len(df) 

221 hgvs_vals = df[hgvs_col].astype(str).tolist() if hgvs_col in df.columns else [""] * len(df) 

222 

223 for gene, hgvs_p in zip(genes, hgvs_vals, strict=True): 

224 parsed = parse_hgvs_p(hgvs_p) 

225 

226 if not parsed or not gene: 

227 results.append(NOT_TRIGGERED.copy()) 

228 continue 

229 

230 ref_aa, aa_pos, alt_aa = parsed 

231 results.append(check_pm5(gene, ref_aa, aa_pos, alt_aa, lookup)) 

232 

233 df = df.copy() 

234 if not results: 

235 df["PM5"] = pd.Series(dtype=str) 

236 df["PM5_evidence_count"] = pd.Series(dtype=int) 

237 df["PM5_known_variants"] = pd.Series(dtype=str) 

238 return df 

239 

240 pm5_df = pd.DataFrame(results) 

241 df["PM5"] = pm5_df["strength"].fillna("").values 

242 df["PM5_evidence_count"] = pm5_df["evidence_count"].fillna(0).astype(int).values 

243 df["PM5_known_variants"] = pm5_df["known_variants"].fillna("").values 

244 return df
