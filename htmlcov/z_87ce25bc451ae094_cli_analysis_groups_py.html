12 # Gene Selection 

13 gene_group = parser.add_argument_group("Gene Selection") 

14 gene_group.add_argument("-g", "--gene-name", help="Gene or list of genes of interest") 

15 gene_group.add_argument("-G", "--gene-file", help="File containing gene names, one per line") 

16 gene_group.add_argument( 

17 "--transcript-list", 

18 help=( 

19 "Comma-separated list of transcript IDs to filter for. " 

20 "For example: NM_007294.4,NM_000059.4" 

21 ), 

22 ) 

23 gene_group.add_argument( 

24 "--transcript-file", 

25 help=( 

26 "Path to a file containing transcript IDs, one per line. " 

27 "For GRCh38, use MANE Select transcript lists matching the VCF namespace " 

28 "(RefSeq NM_* or Ensembl ENST*). For hg19/GRCh37 RefSeq annotations, use " 

29 "a GRCh37 RefSeq Select transcript list because MANE is defined on GRCh38." 

30 ), 

31 ) 

32 gene_group.add_argument( 

33 "--transcript-fallback", 

34 choices=["none", "keep-gene"], 

35 default=None, 

36 help=( 

37 "What to do with a gene none of whose annotated transcripts is in the transcript " 

38 "list. 'none' (default): drop it (reported as a WARNING and, with " 

39 "--diagnostics-output, in transcript_filter_dropped_genes.tsv). 'keep-gene': keep " 

40 "the gene-body annotations of that gene (every transcript becomes a row; readthrough " 

41 "genes without a listed transcript are kept too). keep-gene needs --transcript-list or " 

42 "--transcript-file and runs without parallel chunks." 

43 ), 

44 )

46 "--regions-bed", 

47 help=(

50 "--gene-name all, the regions BED is used directly." 

51 ), 

52 default=None, 

53 )
