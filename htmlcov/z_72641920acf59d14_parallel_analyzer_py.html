26def _get_batch_size(n_workers: int) -> int: 

27 """ 

28 Determine the batch size for parallel gene processing. 

29 

30 Checks ``VARIANTCENTRIFUGE_BATCH_SIZE`` environment variable first. 

31 Falls back to ``min(4 * n_workers, 500)`` if not set or invalid. 

32 

33 Parameters 

34 ---------- 

35 n_workers : int 

36 Number of parallel workers. 

37 

38 Returns 

39 ------- 

40 int 

41 Batch size (number of genes per submission batch). 

42 """ 

43 env_val = os.environ.get("VARIANTCENTRIFUGE_BATCH_SIZE") 

44 if env_val is not None: 

45 try: 

46 batch_size = int(env_val) 

47 if batch_size >= 1: 

48 return batch_size 

49 logger.warning( 

50 f"VARIANTCENTRIFUGE_BATCH_SIZE={env_val!r} must be >= 1; " 

51 "falling back to auto batch size" 

52 ) 

53 except ValueError: 

54 logger.warning( 

55 f"VARIANTCENTRIFUGE_BATCH_SIZE={env_val!r} is not a valid integer; " 

56 "falling back to auto batch size" 

57 ) 

58 return min(4 * n_workers, 500)

61def _process_gene_group( 

62 gene: str, 

63 gene_df: pd.DataFrame, 

64 pedigree_data: dict[str, dict[str, Any]], 

65 sample_list: list[str], 

66 gt_matrix: np.ndarray | None = None, 

67 sample_to_idx: dict[str, int] | None = None, 

68 gene_row_indices: np.ndarray | None = None, 

69 variant_keys: np.ndarray | None = None, 

70) -> tuple[str, dict[str, Any]]: 

71 """ 

72 Process a single gene's variants for compound heterozygous patterns. 

73 

74 This function is designed to be run in a worker thread. 

75 

76 Parameters 

77 ---------- 

78 gene : str 

79 Gene name 

80 gene_df : pd.DataFrame 

81 DataFrame containing variants for this gene 

82 pedigree_data : Dict 

83 Pedigree information 

84 sample_list : List[str] 

85 List of sample IDs 

86 gt_matrix : np.ndarray, optional 

87 Pre-built genotype matrix from Pass 1 (read-only, thread-safe). 

88 sample_to_idx : Dict[str, int], optional 

89 Sample-to-column mapping for gt_matrix. 

90 gene_row_indices : np.ndarray, optional 

91 Row indices into gt_matrix for this gene. 

92 variant_keys : np.ndarray, optional 

93 Pre-computed variant key strings for this gene's rows. 

94 

95 Returns 

96 ------- 

97 Tuple[str, Dict[str, Any]] 

98 Gene name and compound het results 

99 """ 

100 if pd.isna(gene) or gene == "": 

101 return gene, {} 

102 

103 if len(gene_df) <= 1: 

104 return gene, {} 

105 

106 # Use vectorized implementation with optional pre-built matrix 

107 comp_het_results = analyze_gene_for_compound_het_vectorized( 

108 gene_df, 

109 pedigree_data, 

110 sample_list,

112 sample_to_idx=sample_to_idx, 

113 gene_row_indices=gene_row_indices, 

114 variant_keys=variant_keys, 

115 ) 

116 

117 return gene, comp_het_results

120def analyze_inheritance_parallel( 

121 df: pd.DataFrame, 

122 pedigree_data: dict[str, dict[str, Any]], 

123 sample_list: list[str], 

124 use_vectorized_comp_het: bool = True, 

125 n_workers: int | None = None, 

126 min_variants_for_parallel: int = 100, 

127 reference: str | None = None, 

128) -> pd.DataFrame: 

129 """ 

130 Parallel version of inheritance analysis with concurrent gene processing. 

131 

132 This function performs the same analysis as analyze_inheritance but 

133 processes genes in parallel for improved performance on multi-core systems. 

134 

135 Parameters 

136 ---------- 

137 df : pd.DataFrame 

138 DataFrame containing variant data with sample genotypes 

139 pedigree_data : Dict[str, Dict[str, Any]] 

140 Pedigree information dictionary 

141 sample_list : List[str] 

142 List of sample IDs to analyze 

143 use_vectorized_comp_het : bool, optional 

144 Deprecated parameter (kept for backward compatibility). 

145 Vectorized implementation is now always used. 

146 n_workers : int, optional 

147 Number of parallel workers. If None, uses CPU count 

148 min_variants_for_parallel : int, optional 

149 Minimum number of variants to use parallel processing 

150 reference : str, optional 

151 Genome build (e.g. "GRCh37", "hg38") used to place the X PAR regions. 

152 

153 Returns 

154 ------- 

155 pd.DataFrame 

156 DataFrame with added inheritance columns 

157 """ 

158 import time 

159 

160 start_time = time.time() 

161 

162 if df.empty: 

163 logger.warning("Empty DataFrame provided for inheritance analysis") 

164 df["Inheritance_Pattern"] = "none" 

165 df["Inheritance_Details"] = "{}" 

166 return df 

167 

168 # Handle empty pedigree data 

169 if not pedigree_data: 

170 pedigree_data = {} 

171 logger.info("No pedigree data provided - treating samples as unrelated individuals") 

172 

173 logger.info( 

174 f"Starting parallel inheritance analysis for {len(df)} variants " 

175 f"across {len(sample_list)} samples" 

176 ) 

177 

178 # Pre-populate pedigree_data for unknown samples to prevent thread race 

179 # condition in comp_het_vectorized.py (which mutates the dict for missing samples) 

180 for sample_id in sample_list: 

181 if sample_id not in pedigree_data: 

182 pedigree_data[sample_id] = { 

183 "sample_id": sample_id, 

184 "father_id": "0", 

185 "mother_id": "0", 

186 "affected_status": "2", 

187 } 

188 

189 # Create a copy to avoid fragmentation 

190 df = df.copy() 

191 df["_inheritance_patterns"] = None 

192 df["_comp_het_info"] = None 

193 df["Inheritance_Pattern"] = "none" 

194 df["Inheritance_Details"] = "{}" 

195 

196 # Track timing for each pass 

197 pass_times = {} 

198 

199 # Pass 1: Per-Variant Pattern Deduction (vectorized) 

200 # Request genotype matrix back so Pass 2 can reuse it (avoids redundant encoding) 

201 pass1_start = time.time() 

202 logger.info("Pass 1: Deducing inheritance patterns per variant (vectorized)") 

203 pass1_result = vectorized_deduce_patterns( 

204 df, pedigree_data, sample_list, return_genotype_matrix=True, reference=reference 

205 ) 

206 if isinstance(pass1_result, tuple): 

207 patterns_list_all, gt_matrix, sample_to_idx = pass1_result 

208 else: 

209 patterns_list_all = pass1_result 

210 gt_matrix, sample_to_idx = None, None 

211 df["_inheritance_patterns"] = patterns_list_all 

212 pass_times["pattern_deduction"] = time.time() - pass1_start 

213 

214 # Build positional row index array for slicing the genotype matrix per-gene 

215 row_positions = np.arange(len(df)) 

216 

217 # Pre-compute variant keys for all rows (reused per-gene and for result application) 

218 all_variant_keys = build_variant_keys_array(df) 

219 

220 # Pass 2: Compound Heterozygous Analysis (PARALLEL) 

221 pass2_start = time.time() 

222 

223 comp_het_results_by_gene = {} 

224 

225 # Check if we should use parallel processing 

226 from ..memory import ResourceManager 

227 

228 rm = ResourceManager() 

229 

230 # Use min_variants_for_parallel if explicitly provided (for backward compatibility), 

231 # otherwise use ResourceManager's threshold 

232 if min_variants_for_parallel != 100: # 100 is the default 

233 should_parallelize = len(df) >= min_variants_for_parallel 

234 else: 

235 should_parallelize = rm.should_parallelize(len(df)) 

236 

237 use_parallel = ( 

238 (n_workers is None or n_workers > 1) and should_parallelize and "GENE" in df.columns 

239 ) 

240 

241 if use_parallel and "GENE" in df.columns: 

242 # Group by gene 

243 gene_groups = df.groupby("GENE", observed=True) 

244 genes_with_multiple_variants = [] 

245 for gene, group_df in gene_groups: 

246 if len(group_df) > 1 and not pd.isna(gene) and gene != "": 

247 gene_iloc = df.index.get_indexer(group_df.index) 

248 gene_row_idx = row_positions[gene_iloc] 

249 gene_vkeys = all_variant_keys[gene_iloc] 

250 genes_with_multiple_variants.append((gene, group_df, gene_row_idx, gene_vkeys)) 

251 

252 num_genes = len(genes_with_multiple_variants) 

253 if num_genes > 0: 

254 # Sort genes by variant count descending (largest first for load balancing) 

255 genes_with_multiple_variants.sort(key=lambda x: len(x[1]), reverse=True) 

256 max_gene_size = ( 

257 len(genes_with_multiple_variants[0][1]) if genes_with_multiple_variants else 0 

258 ) 

259 

260 # Auto-detect worker count if not specified 

261 if n_workers is None: 

262 memory_per_gene_gb = rm.estimate_memory(max_gene_size, len(sample_list)) 

263 n_workers = rm.auto_workers( 

264 task_count=num_genes, memory_per_task_gb=memory_per_gene_gb 

265 ) 

266 

267 logger.info( 

268 f"Pass 2: Analyzing {num_genes} genes for compound heterozygous patterns " 

269 f"in parallel ({n_workers} workers, largest gene: {max_gene_size} variants)" 

270 ) 

271 

272 # Process genes in parallel using threads (NumPy releases GIL) 

273 batch_size = _get_batch_size(n_workers) 

274 with ThreadPoolExecutor(max_workers=n_workers) as executor: 

275 completed = 0 

276 # Submit in batches to bound concurrent work 

277 for batch_start in range(0, num_genes, batch_size): 

278 batch = genes_with_multiple_variants[batch_start : batch_start + batch_size] 

279 future_to_gene = { 

280 executor.submit( 

281 _process_gene_group, 

282 str(gene), 

283 gene_df, 

284 pedigree_data, 

285 sample_list, 

286 gt_matrix, 

287 sample_to_idx, 

288 gene_row_idx, 

289 gene_vkeys, 

290 ): gene 

291 for gene, gene_df, gene_row_idx, gene_vkeys in batch 

292 } 

293 

294 # Collect results as they complete 

295 for future in as_completed(future_to_gene): 

296 gene = future_to_gene[future] 

297 try: 

298 gene_name, comp_het_results = future.result() 

299 if comp_het_results: 

300 comp_het_results_by_gene[gene_name] = comp_het_results 

301 logger.debug( 

302 f"Found compound het patterns in gene {gene_name}: " 

303 f"{len(comp_het_results)} variants" 

304 ) 

305 completed += 1 

306 if completed % 100 == 0: 

307 logger.info(f"Processed {completed}/{num_genes} genes") 

308 except Exception as e: 

309 logger.error(f"Error processing gene {gene!r}: {e}") 

310 raise 

311 else: 

312 # Fall back to sequential processing 

313 logger.info("Pass 2: Analyzing compound heterozygous patterns sequentially") 

314 

315 if "GENE" in df.columns: 

316 for gene, gene_df in df.groupby("GENE", observed=True): 

317 if pd.isna(gene) or gene == "" or len(gene_df) <= 1: 

318 continue 

319 

320 try: 

321 gene_iloc = df.index.get_indexer(gene_df.index) 

322 gene_row_idx = row_positions[gene_iloc] 

323 gene_vkeys = all_variant_keys[gene_iloc] 

324 

325 comp_het_results = analyze_gene_for_compound_het_vectorized( 

326 gene_df, 

327 pedigree_data, 

328 sample_list, 

329 gt_matrix=gt_matrix, 

330 sample_to_idx=sample_to_idx, 

331 gene_row_indices=gene_row_idx, 

332 variant_keys=gene_vkeys, 

333 ) 

334 

335 if comp_het_results: 

336 comp_het_results_by_gene[str(gene)] = comp_het_results 

337 except Exception as e: 

338 logger.error(f"Error processing gene {gene!r}: {e}") 

339 raise 

340 

341 pass_times["compound_het_analysis"] = time.time() - pass2_start 

342 

343 # Apply compound het results to DataFrame — batch assignment 

344 apply_start = time.time() 

345 if comp_het_results_by_gene: 

346 # Reuse pre-computed variant keys 

347 variant_keys_series = pd.Series(all_variant_keys, index=df.index) 

348 

349 # Get genes array - handle categorical dtype 

350 if isinstance(df["GENE"].dtype, pd.CategoricalDtype): 

351 genes = df["GENE"].astype(str).values 

352 else: 

353 genes = df["GENE"].values.astype(str) 

354 

355 # Collect all (index, info) pairs, then assign 

356 batch_indices: list[int] = [] 

357 batch_values: list[dict] = [] 

358 

359 for gene, gene_results in comp_het_results_by_gene.items(): 

360 gene_mask = genes == gene 

361 if not np.any(gene_mask): 

362 continue 

363 

364 gene_variant_keys = variant_keys_series[gene_mask] 

365 

366 for vk, info in gene_results.items(): 

367 vk_mask = gene_variant_keys == vk 

368 if vk_mask.any(): 

369 matching_indices = gene_variant_keys.index[vk_mask] 

370 for idx in matching_indices: 

371 batch_indices.append(idx) 

372 batch_values.append(info) 

373 

374 # Single batch assignment 

375 if batch_indices: 

376 for i, idx in enumerate(batch_indices): 

377 df.at[idx, "_comp_het_info"] = batch_values[i] 

378 

379 pass_times["apply_comp_het_results"] = time.time() - apply_start 

380 

381 # Pass 3: Prioritize and Finalize 

382 pass3_start = time.time() 

383 logger.info("Pass 3: Prioritizing patterns and creating final output") 

384 

385 _finalize_inheritance_patterns(df, pedigree_data, sample_list) 

386 

387 pass_times["prioritization"] = time.time() - pass3_start 

388 

389 # Clean up temporary columns 

390 df = df.drop(columns=["_inheritance_patterns", "_comp_het_info"]) 

391 

392 # Log summary statistics 

393 total_time = time.time() - start_time 

394 pattern_counts = df["Inheritance_Pattern"].value_counts() 

395 logger.info( 

396 f"Parallel inheritance analysis complete in {total_time:.1f}s. " 

397 f"Pattern distribution: {pattern_counts.to_dict()}" 

398 ) 

399 

400 # Log timing breakdown 

401 logger.debug("Inheritance analysis timing breakdown:") 

402 for pass_name, duration in pass_times.items(): 

403 pct = (duration / total_time) * 100 if total_time > 0 else 0 

404 logger.debug(f" {pass_name}: {duration:.1f}s ({pct:.1f}%)") 

405 

406 return df
